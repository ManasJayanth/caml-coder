def sha(data: bytes) -> str:
    return hashlib.sha256(data).hexdigest()


def canonical(obj: object) -> bytes:
    return json.dumps(obj, sort_keys=True, separators=(",", ":"), ensure_ascii=False,
                      allow_nan=False).encode("utf-8")


def cache_inventory_equal(left: list[dict], right: list[dict]) -> bool:
    """Ignore only cache-root directory mtime; retain full inventories for diagnostics."""
    def semantic(rows: list[dict]) -> list[dict]:
        return [{k: v for k, v in row.items()
                 if not (row.get("path") == "." and row.get("type") == "dir" and k == "mtime_ms")}
                for row in rows]
    return canonical(semantic(left)) == canonical(semantic(right))


def read_bounded(path: Path, limit: int) -> bytes:
    st = path.lstat()
    if not path.is_file() or path.is_symlink() or st.st_size > limit:
        raise ValueError(f"input must be a regular file no larger than {limit} bytes")
    with path.open("rb") as stream:
        data = stream.read(limit + 1)
    if len(data) > limit:
        raise ValueError(f"input exceeds {limit} byte limit")
    return data


def bounded_process(argv: list[str], timeout: float, cap: int = MAX_DOCKER_BYTES,
                    stdin: bytes | None = None, *, deadline: float | None = None,
                    execution_deadline: float | None = None) -> dict:
    """Drain both pipes and bounded stdin under one monotonic deadline."""
    if os.name != "posix":
        raise RuntimeError("runner requires POSIX process-group supervision")
    deadline = min(deadline, time.monotonic() + timeout) if deadline is not None else time.monotonic() + timeout
    run_deadline = min(deadline, execution_deadline) if execution_deadline is not None else deadline
    if run_deadline <= time.monotonic():
        return {"exit_code": None, "timed_out": True, "output_exceeded": False,
                "interrupted": False, "group_cleanup_uncertain": False,
                "total_bytes": 0, "stdout_total_bytes": 0, "stderr_total_bytes": 0,
                "stdout_full_sha256": hashlib.sha256(b"").hexdigest(),
                "stderr_full_sha256": hashlib.sha256(b"").hexdigest(), "stdout": b"", "stderr": b""}
    proc = subprocess.Popen(argv, stdin=subprocess.PIPE if stdin is not None else subprocess.DEVNULL,
                            stdout=subprocess.PIPE, stderr=subprocess.PIPE, start_new_session=True,
                            close_fds=True)
    streams = {"stdout": bytearray(), "stderr": bytearray()}
    stream_hashes = {"stdout": hashlib.sha256(), "stderr": hashlib.sha256()}
    stream_sizes = {"stdout": 0, "stderr": 0}
    total = 0
    overflow = False
    sent = 0
    timed_out = False
    sel = None
    interrupted = False
    group_cleanup_uncertain = False
    needs_group_cleanup = True

    def signal_group(sig: int) -> bool:
        nonlocal group_cleanup_uncertain
        try:
            os.killpg(proc.pid, sig)
            return True
        except ProcessLookupError:
            return False
        except OSError:
            group_cleanup_uncertain = True
            return False

    def stop_group() -> None:
        nonlocal group_cleanup_uncertain
        left = max(0.0, deadline - time.monotonic())
        if left <= 0:
            group_cleanup_uncertain = True
            signal_group(signal.SIGKILL)
            return
        signal_group(signal.SIGTERM)
        grace = min(.2, max(0.0, deadline - time.monotonic()))
        if grace:
            try: proc.wait(timeout=grace)
            except subprocess.TimeoutExpired: pass
        # Always signal the group: the leader may have exited while a descendant
        # still owns a pipe or continues running.
        signal_group(signal.SIGKILL)
        left = max(0.0, deadline - time.monotonic())
        try: proc.wait(timeout=left)
        except subprocess.TimeoutExpired: group_cleanup_uncertain = True
        while True:
            remaining = deadline - time.monotonic()
            if remaining <= 0:
                break
            if not signal_group(0):
                break
            time.sleep(min(0.02, remaining))
        if signal_group(0):
            group_cleanup_uncertain = True

    try:
        sel = selectors.DefaultSelector()
        if stdin is not None:
            os.set_blocking(proc.stdin.fileno(), False)
            sel.register(proc.stdin, selectors.EVENT_WRITE, "stdin")
        for label, pipe in (("stdout", proc.stdout), ("stderr", proc.stderr)):
            os.set_blocking(pipe.fileno(), False)
            sel.register(pipe, selectors.EVENT_READ, label)
        while sel.get_map():
            left = run_deadline - time.monotonic()
            if left <= 0:
                timed_out = True
                break
            for key, _ in sel.select(min(left, .1)):
                if key.data == "stdin":
                    if sent == len(stdin):
                        sel.unregister(key.fileobj); key.fileobj.close(); continue
                    try:
                        sent += os.write(key.fd, stdin[sent:sent + 65536])
                    except BrokenPipeError:
                        sel.unregister(key.fileobj); key.fileobj.close()
                else:
                    block = os.read(key.fd, 65536)
                    if not block:
                        sel.unregister(key.fileobj); continue
                    total += len(block)
                    stream_sizes[key.data] += len(block)
                    stream_hashes[key.data].update(block)
                    room = max(0, cap - sum(map(len, streams.values())))
                    streams[key.data].extend(block[:room])
                    if total > cap:
                        overflow = True
                        break
            if overflow:
                break
        if timed_out or overflow:
            stop_group()
            needs_group_cleanup = False
        else:
            try:
                proc.wait(timeout=max(0, run_deadline - time.monotonic()))
            except subprocess.TimeoutExpired:
                timed_out = True
            # The group remains owned after its leader exits, even after EOF.
            stop_group()
            needs_group_cleanup = False
    except KeyboardInterrupt:
        interrupted = True
        stop_group()
        needs_group_cleanup = False
    finally:
        if needs_group_cleanup:
            try:
                stop_group()
            except OSError:
                group_cleanup_uncertain = True
        if sel is not None:
            sel.close()
        for pipe in (proc.stdin, proc.stdout, proc.stderr):
            if pipe is not None:
                try: pipe.close()
                except OSError: pass
    return {"exit_code": proc.returncode, "timed_out": timed_out, "output_exceeded": overflow,
            "interrupted": interrupted, "group_cleanup_uncertain": group_cleanup_uncertain,
            "total_bytes": total, "stdout_total_bytes": stream_sizes["stdout"],
            "stderr_total_bytes": stream_sizes["stderr"],
            "stdout_full_sha256": stream_hashes["stdout"].hexdigest(),
            "stderr_full_sha256": stream_hashes["stderr"].hexdigest(),
            "stdout": bytes(streams["stdout"]), "stderr": bytes(streams["stderr"])}

def docker_image(image: str) -> dict:
    p = bounded_process(["docker", "image", "inspect", image], 15)
    if p["exit_code"] != 0 or p["timed_out"] or p["output_exceeded"] or p["interrupted"] or p["group_cleanup_uncertain"]:
        raise RuntimeError("bounded image inspection failed")
    rows = json.loads(p["stdout"])
    if len(rows) != 1 or rows[0].get("Id") != image:
        raise ValueError("image must resolve to its exact immutable ID")
    return rows[0]

def docker_container_inspect(container_id: str, *, deadline: float | None = None) -> tuple[str, dict | None]:
    p = bounded_process(["docker", "container", "inspect", container_id], 10, deadline=deadline)
    if p["timed_out"] or p["output_exceeded"] or p["interrupted"] or p["group_cleanup_uncertain"]:
        return "query-failed", None
    if p["exit_code"]:
        # A successful daemon query with an exact no-such-container response is absence.
        err = p["stderr"].decode("utf-8", "replace").lower()
        if "no such container" not in err: return "query-failed", None
        filter_key = "id" if re.fullmatch(r"[0-9a-f]{64}", container_id) else "name"
        listing = bounded_process(["docker", "container", "ls", "-a", "--no-trunc",
                                   "--filter", filter_key+"="+container_id,
                                   "--format", "{{.Names}}" if filter_key == "name" else "{{.ID}}"], 10, deadline=deadline)
        if listing["exit_code"] or listing["timed_out"] or listing["output_exceeded"] or listing["interrupted"] or listing["group_cleanup_uncertain"]:
            return "query-failed", None
        rows = listing["stdout"].decode("utf-8", "replace").splitlines()
        exists = any(container_id in row.split(",") for row in rows) if filter_key == "name" else bool(rows)
        return ("query-failed", None) if exists else ("absent", None)
    try:
        rows = json.loads(p["stdout"])
        return ("present", rows[0]) if len(rows) == 1 else ("query-failed", None)
    except (ValueError, TypeError):
        return "query-failed", None


def cleanup_owned(container_id: str | None, owner: str, *, name: str | None = None,
                  deadline: float | None = None, previously_verified: bool = False) -> dict:
    # A create can succeed even when its client times out. Recover only from the
    # persisted unique name after checking both Docker's name and owner label.
    if not container_id and name:
        state, obj = docker_container_inspect(name, deadline=deadline)
        if state == "present":
            labels = obj.get("Config", {}).get("Labels", {}) or {}
            actual_name = (obj.get("Name") or "").lstrip("/")
            candidate = obj.get("Id")
            if actual_name == name and labels.get("esy-reference.owner") == owner and re.fullmatch(r"[0-9a-f]{64}", str(candidate)):
                container_id = candidate
                previously_verified = True
        elif state != "absent":
            return {"verified_absent": False, "reason": "recovery-query-failed"}
        else:
            return {"verified_absent": True, "reason": "daemon-confirmed-absent"}
    if not container_id:
        return {"verified_absent": False, "reason": "container-id-unavailable"}
    if not previously_verified:
        state, obj = docker_container_inspect(container_id, deadline=deadline)
        if state == "absent": return {"verified_absent": True, "reason": "daemon-confirmed-absent"}
        if state != "present": return {"verified_absent": False, "reason": "inspect-query-failed"}
        labels = obj.get("Config", {}).get("Labels", {}) or {}
        actual_name = (obj.get("Name") or "").lstrip("/")
        if (obj.get("Id") != container_id or labels.get("esy-reference.owner") != owner or
                (name is not None and actual_name != name)):
            return {"verified_absent": False, "reason": "ownership-mismatch"}
    left = 10.0 if deadline is None else max(0.0, deadline-time.monotonic())
    if left <= 0: return {"verified_absent": False, "reason": "cleanup-deadline-exhausted"}
    rm = bounded_process(["docker", "rm", "-f", container_id], min(10.0, max(.001, left*.65)), deadline=deadline)
    if rm["timed_out"] or rm["interrupted"] or rm["group_cleanup_uncertain"]:
        return {"remove_exit_code": rm["exit_code"], "verified_absent": False, "reason": "remove-uncertain"}
    state2, _ = docker_container_inspect(container_id, deadline=deadline)
    return {"remove_exit_code": rm["exit_code"], "verified_absent": state2 == "absent",
            "reason": "removed-and-daemon-confirmed-absent" if state2 == "absent" else "absence-unverified"}
