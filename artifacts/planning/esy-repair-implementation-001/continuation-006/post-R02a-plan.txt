def cache_inventory_equal(left: list[dict], right: list[dict]) -> bool:
    """Ignore only cache-root directory mtime; retain full inventories for diagnostics."""
    def semantic(rows: list[dict]) -> list[dict]:
        return [{k: v for k, v in row.items()

def read_bounded(path: Path, limit: int) -> bytes:
    st = path.lstat()
    if not path.is_file() or path.is_symlink() or st.st_size > limit:
        raise ValueError(f"input must be a regular file no larger than {limit} bytes")
    with path.open("rb") as stream:
        data = stream.read(limit + 1)
    if len(data) > limit:
        raise ValueError(f"input exceeds {limit} byte limit")
    return data


def bounded_process(argv: list[str], timeout: float, cap: int = MAX_DOCKER_BYTES,
                    stdin: bytes | None = None, *, deadline: float | None = None,
                    execution_deadline: float | None = None) -> dict:
    """Drain both pipes and bounded stdin under one monotonic deadline."""
    if os.name != "posix":
        raise RuntimeError("runner requires POSIX process-group supervision")
    deadline = min(deadline, time.monotonic() + timeout) if deadline is not None else time.monotonic() + timeout
    run_deadline = min(deadline, execution_deadline) if execution_deadline is not None else deadline
    if run_deadline <= time.monotonic():
        return {"exit_code": None, "timed_out": True, "output_exceeded": False,
                "interrupted": False, "group_cleanup_uncertain": False,
                "total_bytes": 0, "stdout_total_bytes": 0, "stderr_total_bytes": 0,

    stream_hashes = {"stdout": hashlib.sha256(), "stderr": hashlib.sha256()}
    stream_sizes = {"stdout": 0, "stderr": 0}
    total = 0
    overflow = False
    sent = 0

    interrupted = False
    group_cleanup_uncertain = False
    needs_group_cleanup = True

        nonlocal group_cleanup_uncertain
        try:
            os.killpg(proc.pid, sig)
            return True
        except ProcessLookupError:
            return False
        except OSError:
            group_cleanup_uncertain = True
            return False

        nonlocal group_cleanup_uncertain
        left = max(0.0, deadline - time.monotonic())

            group_cleanup_uncertain = True
            signal_group(signal.SIGKILL)

def docker_image(image: str) -> dict:
    p = bounded_process(["docker", "image", "inspect", image], 15)
    if p["exit_code"] != 0 or p["timed_out"] or p["output_exceeded"] or p["interrupted"] or p["group_cleanup_uncertain"]:
        raise RuntimeError("bounded image inspection failed")
    rows = json.loads(p["stdout"])
    if len(rows) != 1 or rows[0].get("Id") != image:
        raise ValueError("image must resolve to its exact immutable ID")
    return rows[0]

def docker_container_inspect(container_id: str, *, deadline: float | None = None) -> tuple[str, dict | None]:
    p = bounded_process(["docker", "container", "inspect", container_id], 10, deadline=deadline)
    if p["timed_out"] or p["output_exceeded"] or p["interrupted"] or p["group_cleanup_uncertain"]:
        return "query-failed", None
    if p["exit_code"]:
        # A successful daemon query with an exact no-such-container response is absence.
        err = p["stderr"].decode("utf-8", "replace").lower()
        if "no such container" not in err: return "query-failed", None
        filter_key = "id" if re.fullmatch(r"[0-9a-f]{64}", container_id) else "name"
        listing = bounded_process(["docker", "container", "ls", "-a", "--no-trunc",
                                   "--filter", filter_key+"="+container_id,
                                   "--format", "{{.Names}}" if filter_key == "name" else "{{.ID}}"], 10, deadline=deadline)
        if listing["exit_code"] or listing["timed_out"] or listing["output_exceeded"] or listing["interrupted"] or listing["group_cleanup_uncertain"]:
            return "query-failed", None
        rows = listing["stdout"].decode("utf-8", "replace").splitlines()
        exists = any(container_id in row.split(",") for row in rows) if filter_key == "name" else bool(rows)
        return ("query-failed", None) if exists else ("absent", None)
    try:
        rows = json.loads(p["stdout"])
        return ("present", rows[0]) if len(rows) == 1 else ("query-failed", None)
    except (ValueError, TypeError):
        return "query-failed", None


def cleanup_owned(container_id: str | None, owner: str, *, name: str | None = None,
                  deadline: float | None = None, previously_verified: bool = False) -> dict:
    # A create can succeed even when its client times out. Recover only from the
    # persisted unique name after checking both Docker's name and owner label.
    if not container_id and name:
        state, obj = docker_container_inspect(name, deadline=deadline)
        if state == "present":
            labels = obj.get("Config", {}).get("Labels", {}) or {}
            actual_name = (obj.get("Name") or "").lstrip("/")
            candidate = obj.get("Id")
            if actual_name == name and labels.get("esy-reference.owner") == owner and re.fullmatch(r"[0-9a-f]{64}", str(candidate)):
                container_id = candidate
                previously_verified = True
        elif state != "absent":
            return {"verified_absent": False, "reason": "recovery-query-failed"}
        else:
            return {"verified_absent": True, "reason": "daemon-confirmed-absent"}
    if not container_id:
        return {"verified_absent": False, "reason": "container-id-unavailable"}
    if not previously_verified:
        state, obj = docker_container_inspect(container_id, deadline=deadline)
        if state == "absent": return {"verified_absent": True, "reason": "daemon-confirmed-absent"}
        if state != "present": return {"verified_absent": False, "reason": "inspect-query-failed"}
        labels = obj.get("Config", {}).get("Labels", {}) or {}
        actual_name = (obj.get("Name") or "").lstrip("/")
        if (obj.get("Id") != container_id or labels.get("esy-reference.owner") != owner or
                (name is not None and actual_name != name)):
            return {"verified_absent": False, "reason": "ownership-mismatch"}
    left = 10.0 if deadline is None else max(0.0, deadline-time.monotonic())
    if left <= 0: return {"verified_absent": False, "reason": "cleanup-deadline-exhausted"}
    rm = bounded_process(["docker", "rm", "-f", container_id], min(10.0, max(.001, left*.65)), deadline=deadline)
    if rm["timed_out"] or rm["interrupted"] or rm["group_cleanup_uncertain"]:
        return {"remove_exit_code": rm["exit_code"], "verified_absent": False, "reason": "remove-uncertain"}
    state2, _ = docker_container_inspect(container_id, deadline=deadline)
    return {"remove_exit_code": rm["exit_code"], "verified_absent": state2 == "absent",
            "reason": "removed-and-daemon-confirmed-absent" if state2 == "absent" else "absence-unverified"}

def validate_config(obj: dict, image: str, owner: str, mode: str) -> None:
    c, h = obj.get("Config", {}), obj.get("HostConfig", {})
    if obj.get("Image") != image or c.get("User") != "1000:1000" or c.get("WorkingDir") != "/home/opam/esy-reference":
        raise ValueError("container identity/user/workdir mismatch")
    if (c.get("Labels", {}) or {}).get("esy-reference.owner") != owner:
        raise ValueError("container ownership label mismatch")
    if h.get("NetworkMode") != "none" or h.get("ReadonlyRootfs") is not True or h.get("Memory") != 2*1024**3 or h.get("NanoCpus") != 2_000_000_000 or h.get("PidsLimit") != 128:
        raise ValueError("effective resource configuration mismatch")
    if h.get("CapDrop") != ["ALL"] or h.get("SecurityOpt") != ["no-new-privileges:true"]:
        raise ValueError("effective security configuration mismatch")
    if (h.get("Privileged") is not False or h.get("CapAdd") not in (None, []) or
            h.get("Devices") not in (None, []) or h.get("DeviceRequests") not in (None, []) or
            h.get("PidMode") not in (None, "") or h.get("IpcMode") != "private"):
        raise ValueError("effective privilege/device/PID/IPC configuration mismatch")
    env = c.get("Env")
    tmpdir = [entry for entry in env or [] if isinstance(entry, str) and entry.split("=", 1)[0] == "TMPDIR"]
    if not isinstance(env, list) or len(tmpdir) != 1 or tmpdir[0] != "TMPDIR=/work":
        raise ValueError("effective TMPDIR environment mismatch")
    if c.get("Entrypoint") != ["/home/opam/node/bin/node"] or c.get("Cmd") != ["-e", NODE, mode]:
        raise ValueError("effective entrypoint/command mismatch")
    if c.get("Volumes"):
        raise ValueError("image declares unexpected volumes")
    expected = {"/work":"rw,nosuid,nodev,exec,size=128m,uid=1000,gid=1000,mode=700"}
    if mode == "run": expected.update({"/home/opam/esy-reference/_esy":"rw,nosuid,nodev,exec,size=32m,uid=1000,gid=1000,mode=700",
                                        "/home/opam/esy-reference/node_modules/.cache":"rw,nosuid,nodev,exec,size=32m,uid=1000,gid=1000,mode=700"})
    if h.get("Tmpfs") != expected:
        raise ValueError("effective tmpfs destinations/options mismatch")
    mounts = obj.get("Mounts")
    destinations = []
    if not isinstance(mounts, list):
        raise ValueError("effective mount set mismatch")
    for mount in mounts:
        if (not isinstance(mount, dict) or mount.get("Type") != "tmpfs" or
                mount.get("Destination") not in expected or mount.get("RW") is not True or
                mount.get("Source") not in (None, "") or mount.get("Name") not in (None, "") or
                mount.get("Driver") not in (None, "") or mount.get("Propagation") not in (None, "")):
            raise ValueError("effective mount set mismatch")
        destinations.append(mount["Destination"])
    if len(destinations) != len(set(destinations)):
        raise ValueError("effective mount set mismatch")

def atomic_json(path: Path, value: object) -> None:
    path.parent.mkdir(parents=True,exist_ok=True); data=canonical(value)+b"\n"
    fd,tmp=tempfile.mkstemp(prefix=".esyref-",dir=path.parent)
    try:
        with os.fdopen(fd,"wb") as f: f.write(data); f.flush(); os.fsync(f.fileno())
        os.link(tmp,path); d=os.open(path.parent,os.O_RDONLY)
        try: os.fsync(d)
        finally: os.close(d)
    finally: os.unlink(tmp)


def persist_invocation(path: Path, name: str, owner: str, container_id: str | None) -> None:
    """Keep the unique recovery identity in the caller's explicit evidence tree."""
    value = {"schema": 1, "name": name, "owner": owner, "container_id": container_id}
    if container_id is None:
        path.parent.mkdir(parents=True, exist_ok=True)
        fd = os.open(path, os.O_CREAT | os.O_EXCL | os.O_WRONLY, 0o600)
        with os.fdopen(fd, "wb") as stream:
            stream.write(canonical(value) + b"\n"); stream.flush(); os.fsync(stream.fileno())
        directory = os.open(path.parent, os.O_RDONLY)
        try: os.fsync(directory)
        finally: os.close(directory)
    else:
        tmp = path.with_name(path.name + ".tmp-" + secrets.token_hex(4))
        fd = os.open(tmp, os.O_CREAT | os.O_EXCL | os.O_WRONLY, 0o600)
        try:
            with os.fdopen(fd, "wb") as stream:
                stream.write(canonical(value) + b"\n"); stream.flush(); os.fsync(stream.fileno())
            os.replace(tmp, path)
        finally:
            try: tmp.unlink()
            except FileNotFoundError: pass

def evaluate(case: str, result: dict) -> tuple[str,str]:

def require(ok: bool, message: str) -> None:

def canonical(obj: object) -> bytes:
    return json.dumps(obj, sort_keys=True, separators=(",", ":"), ensure_ascii=False,
                      allow_nan=False).encode("utf-8")


def bounded_read(path: Path, limit: int) -> bytes:
    st = path.lstat()
    require(path.is_file() and not path.is_symlink(), f"{path} must be a regular non-symlink file")
    require(st.st_size <= limit, f"{path} exceeds {limit} bytes")
    with path.open("rb") as stream:
        raw = stream.read(limit + 1)
    require(len(raw) <= limit, f"{path} grew beyond {limit} bytes while reading")
    return raw


def digest(data: bytes) -> str:
    return hashlib.sha256(data).hexdigest()


def immutable_write(path: Path, data: bytes) -> str:
    path.parent.mkdir(parents=True, exist_ok=True)
    fd = os.open(path, os.O_WRONLY | os.O_CREAT | os.O_EXCL, 0o600)
    with os.fdopen(fd, "wb") as stream:
        stream.write(data); stream.flush(); os.fsync(stream.fileno())
    directory = os.open(path.parent, os.O_RDONLY)
    try: os.fsync(directory)
    finally: os.close(directory)
    return digest(data)

            "stdout_full_sha256": captures["stdout"]["hash"].hexdigest(), "stdout_retained_sha256": digest(out),
            "stdout_path": str(stdout_path), "stderr_total_bytes": captures["stderr"]["size"],
            "stderr_retained_bytes": len(err), "stderr_full_sha256": captures["stderr"]["hash"].hexdigest(),
            "stderr_retained_sha256": digest(err), "stderr_path": str(stderr_path)}


def suite_cleanup_owned(output: Path, run_id: str, cwd: Path, evidence: Path) -> dict:
    """Independently clean only the exact invocation recorded by the runner."""
    cleanup_deadline = time.monotonic() + 12
    record_path = output / "runner-reports" / "owned-invocations" / (run_id + ".json")
    try: record = json.loads(bounded_read(record_path, 4096))
    except (OSError, E2EFailure, json.JSONDecodeError) as exc:
        return {"verified_absent": False, "reason": f"ownership-record-unavailable: {exc}"}
    if not isinstance(record, dict): return {"verified_absent": False, "reason": "ownership-record-invalid"}
    name, owner = record.get("name"), record.get("owner")
    if record.get("schema") != 1 or re.fullmatch(r"esyref-[0-9a-f]{24}", str(name)) is None or re.fullmatch(r"[0-9a-f]{32}", str(owner)) is None:
        return {"verified_absent": False, "reason": "ownership-record-invalid"}
    if time.monotonic() >= cleanup_deadline: return {"verified_absent": False, "reason": "suite-cleanup-deadline-expired"}
    inspect = supervisor(["docker", "container", "inspect", name], cwd, cleanup_deadline-time.monotonic(),
                         evidence / "owned-inspect.stdout", evidence / "owned-inspect.stderr", deadline=cleanup_deadline)
    if inspect["interrupted"] or inspect["timed_out"] or inspect["process_group_cleanup_uncertain"]:
        return {"verified_absent": False, "reason": "owned-inspect-uncertain"}
    if inspect["exit_code"]:
        err = bounded_read(Path(inspect["stderr_path"]), MAX_LOG).decode("utf-8", "replace").lower()
        if "no such container" not in err: return {"verified_absent": False, "reason": "owned-inspect-failed"}
        if time.monotonic() >= cleanup_deadline: return {"verified_absent": False, "reason": "suite-cleanup-deadline-expired"}
        listing = supervisor(["docker", "container", "ls", "-a", "--no-trunc", "--filter", "name="+name,
                              "--format", "{{.Names}}"], cwd, cleanup_deadline-time.monotonic(),
                             evidence / "owned-list.stdout", evidence / "owned-list.stderr", deadline=cleanup_deadline)
        if listing["exit_code"] or listing["timed_out"] or listing["interrupted"] or listing["process_group_cleanup_uncertain"]:
            return {"verified_absent": False, "reason": "daemon-list-query-failed"}
        names = bounded_read(Path(listing["stdout_path"]), MAX_LOG).decode("utf-8", "replace").splitlines()
        absent = not any(name in row.split(",") for row in names)
        return {"verified_absent": absent, "reason": "daemon-confirmed-absent" if absent else "container-name-still-listed"}
    try: rows = json.loads(bounded_read(Path(inspect["stdout_path"]), MAX_LOG))
    except (json.JSONDecodeError, E2EFailure): return {"verified_absent": False, "reason": "owned-inspect-invalid"}
    if not isinstance(rows, list) or len(rows) != 1: return {"verified_absent": False, "reason": "owned-inspect-invalid"}
    row = rows[0]; labels = (row.get("Config", {}).get("Labels", {}) or {})
    cid = row.get("Id")
    if (row.get("Name", "").lstrip("/") != name or labels.get("esy-reference.owner") != owner or
            re.fullmatch(r"[0-9a-f]{64}", str(cid)) is None or
            record.get("container_id") not in (None, cid)):
        return {"verified_absent": False, "reason": "owned-inspect-ownership-mismatch"}
    if time.monotonic() >= cleanup_deadline: return {"verified_absent": False, "reason": "suite-cleanup-deadline-expired"}
    removed = supervisor(["docker", "rm", "-f", cid], cwd, cleanup_deadline-time.monotonic(),
                         evidence / "owned-remove.stdout", evidence / "owned-remove.stderr", deadline=cleanup_deadline)
    if removed["interrupted"] or removed["timed_out"] or removed["process_group_cleanup_uncertain"]:
        return {"verified_absent": False, "reason": "owned-remove-uncertain"}
    if time.monotonic() >= cleanup_deadline: return {"verified_absent": False, "reason": "suite-cleanup-deadline-expired"}
    verify = supervisor(["docker", "container", "inspect", cid], cwd, cleanup_deadline-time.monotonic(),
                        evidence / "owned-verify.stdout", evidence / "owned-verify.stderr", deadline=cleanup_deadline)
    if verify["interrupted"] or verify["timed_out"] or verify["process_group_cleanup_uncertain"]:
        return {"verified_absent": False, "reason": "owned-verification-uncertain"}
    if verify["exit_code"]:
        err = bounded_read(Path(verify["stderr_path"]), MAX_LOG).decode("utf-8", "replace").lower()
        if "no such container" not in err: return {"verified_absent": False, "reason": "absence-unverified"}
        if time.monotonic() >= cleanup_deadline: return {"verified_absent": False, "reason": "suite-cleanup-deadline-expired"}
        listing = supervisor(["docker", "container", "ls", "-a", "--no-trunc", "--filter", "id="+cid,
                              "--format", "{{.ID}}"], cwd, cleanup_deadline-time.monotonic(),
                             evidence / "owned-verify-list.stdout", evidence / "owned-verify-list.stderr", deadline=cleanup_deadline)
        if listing["exit_code"] or listing["timed_out"] or listing["interrupted"] or listing["process_group_cleanup_uncertain"]:
            return {"verified_absent": False, "reason": "daemon-list-query-failed"}
        ids = bounded_read(Path(listing["stdout_path"]), MAX_LOG).decode("utf-8", "replace").splitlines()
        absent = not ids
        return {"verified_absent": absent, "reason": "daemon-confirmed-absent" if absent else "container-id-still-listed"}
    return {"verified_absent": False, "reason": "container-still-present"}


def docker_inspect(image: str, cwd: Path, evidence: Path) -> dict:
    argv = ["docker", "image", "inspect", image]
    row = supervisor(argv, cwd, 20, evidence / "image-inspect.stdout", evidence / "image-inspect.stderr")
    require(row["exit_code"] == 0 and not row["timed_out"], "real Docker image inspection failed")
    require(row["stdout_total_bytes"] <= MAX_LOG and row["stderr_total_bytes"] <= MAX_LOG,
            "Docker image metadata or diagnostics exceeded bounded capture")
    raw = bounded_read(Path(row["stdout_path"]), MAX_LOG)
    # The retained log is intentionally bounded; metadata beyond the cap fails.
    rows = json.loads(raw)
    require(isinstance(rows, list) and len(rows) == 1, "Docker did not return exactly one image")
    obj = rows[0]
    require(obj.get("Id") == image, "Docker image ID differs from --image")
    require(not obj.get("Config", {}).get("Volumes"), "Docker image declares unexpected volumes")
    layers = obj.get("RootFS", {}).get("Layers")
    require(isinstance(layers, list) and layers, "Docker image has no layer inventory")
    return {"id": obj["Id"], "layers": layers, "config": obj.get("Config", {}),
            "architecture": obj.get("Architecture"), "os": obj.get("Os"), "inspection": row}


def check_expectations(raw: bytes, image: str, runner: Path, dockerfile: Path,
                       fixture: Path) -> dict:
    value = json.loads(raw)
    require(isinstance(value, dict) and value.get("schema_version") == 1,
            "expectations must use captured schema_version 1")
    require(value.get("image") == image, "expectation image must equal --image")
    require(value.get("base_image") == "sha256:c41892ac54ffae381d3629618f07134e8191a609baec19a77584f45c5345c267",
            "expectations do not pin the required base image")
    require(isinstance(value.get("base_layers"), list) and value["base_layers"], "base layers missing")
    require(isinstance(value.get("image_layers"), list) and value["image_layers"], "image layers missing")
    require(value["image_layers"][:len(value["base_layers"])] == value["base_layers"],
            "captured derivative does not retain base layer prefix")
    src = value.get("source_sha256")
    required = {"run_esy_reference.py": runner, "Dockerfile.esy-runtime": dockerfile, "main.ml": fixture}
    require(isinstance(src, dict), "captured source hash inventory missing")
    for name, path in required.items():
        expected = src.get(name)
        require(isinstance(expected, str) and SHA_RE.fullmatch(expected), f"captured hash missing: {name}")
        require(digest(bounded_read(path, 1024 * 1024)) == expected, f"reviewed source hash mismatch: {name}")
    return value


def run_case(name: str, case: str, timeout: int, expect_pass: bool, runner: Path,
             image: str, expectations: Path, output: Path, cwd: Path) -> dict:
    case_dir = output / "real-runs" / name
    case_dir.mkdir(parents=True, exist_ok=False)
    run_id = "e2e-" + name
    argv = [sys.executable, "-B", str(runner), "--image", image, "--run-id", run_id,
            "--expectations", str(expectations), "--output-dir", str(output / "runner-reports"),
            "--case", case, "--timeout", str(timeout)]
    supervision_result = None
    supervision_error = None
    independent_cleanup_result = None
    independent_cleanup_error = None
    if name == "cleanup-finally-log-write":
        (case_dir / "runner.stdout").mkdir()
    try:
        supervision_result = supervisor(argv, cwd, timeout + 30, case_dir / "runner.stdout", case_dir / "runner.stderr")
    except BaseException as exc:
        supervision_error = {"type": type(exc).__name__, "message": str(exc)}
    finally:
        try:
            independent_cleanup_result = suite_cleanup_owned(output, run_id, cwd, case_dir)
        except BaseException as exc:
            independent_cleanup_error = {"type": type(exc).__name__, "message": str(exc)}
    proc = supervision_result
    suite_cleanup = independent_cleanup_result
    if supervision_error is not None or independent_cleanup_error is not None:
        raise ScenarioFailure({"supervision_result": supervision_result, "supervision_error": supervision_error,
                               "independent_cleanup_result": independent_cleanup_result,
                               "independent_cleanup_error": independent_cleanup_error})
    require(suite_cleanup.get("verified_absent") is True,
            f"independent owned-container cleanup failed for {name}: {suite_cleanup.get('reason')}")
    require(not proc["interrupted"], f"runner supervision was interrupted for {name}")
    require(not proc["process_group_cleanup_uncertain"], f"runner process-group cleanup is uncertain for {name}")
    raw = bounded_read(Path(proc["stdout_path"]), MAX_LOG)
    try: cli = json.loads(raw)
    except (UnicodeDecodeError, json.JSONDecodeError) as exc: raise E2EFailure(f"{name}: malformed runner CLI output") from exc
    report = None
    failures = []
    if not isinstance(cli, dict) or not isinstance(cli.get("report"), str):
        failures.append("runner did not return a report path")
    else:
        report_path = Path(cli["report"])
        require(report_path.resolve().parent == (output / "runner-reports").resolve(),
                "runner report escaped its owned report directory")
        report_raw = bounded_read(report_path, MAX_REPORT)
        report = json.loads(report_raw)
        immutable_write(case_dir / "runner-report.json", report_raw)
        require(report.get("id") == run_id, "runner report ID mismatch")
        ownership_path = output / "runner-reports" / "owned-invocations" / (run_id + ".json")
        ownership = json.loads(bounded_read(ownership_path, 4096))
        require(ownership.get("schema") == 1 and isinstance(ownership.get("name"), str) and
                re.fullmatch(r"esyref-[0-9a-f]{24}", ownership["name"]) is not None and
                re.fullmatch(r"[0-9a-f]{32}", str(ownership.get("owner"))) is not None,
                "runner ownership record is malformed")
        require(ownership.get("container_id") == report.get("container_id"),
                "runner ownership record and report container IDs differ")
        require(report.get("interrupted") is not True, "runner report records interruption")
        require(report.get("process_group_cleanup_uncertain") is not True, "runner report records process-group cleanup uncertainty")
        require(report.get("cleanup", {}).get("verified_absent") is True,
                "runner lacks daemon-confirmed container cleanup")
    require(not proc["timed_out"], f"host runner exceeded its bounded deadline for {name}")
    if expect_pass:
        require(proc["exit_code"] == 0 and report and report.get("status") == "PASSED",
                f"{name}: positive invocation did not exit zero with PASSED report")
        if case in ("positive", "suite"):
            require(report.get("fixture_hashes", {}).get("main.ml") == digest(bounded_read(runner.parent / "main.ml", 64*1024)),
                    "reference main.ml bytes changed in runner payload")
    else:
        require(proc["exit_code"] != 0 and report and report.get("status") == "FAILED",
                f"{name}: negative control did not return nonzero with FAILED report")
        reason = report.get("reason")
        required_reason = {"wrong-result":"wrong-result-observed","syntax":"syntax-failure-observed",
                           "timeout":"timeout-observed","output":"output-observed",
                           "resource":"resource-allocation-failure-observed"}.get(case)
        if required_reason:
            allowed = ("resource-allocation-failure-observed", "resource-oom-observed") if case == "resource" else (required_reason,)
            require(reason in allowed, f"{name}: wrong observed reason {reason!r}")
            if case == "resource" and reason == "resource-allocation-failure-observed":
                evidence = report.get("results", {}).get("commands", {}).get("run-resource", {})
                require(evidence.get("reason") == "allocation-failure" and "Out of memory" in evidence.get("stderr", ""),
                        "resource failure lacks an OCaml allocation diagnostic")
            if case == "resource" and reason == "resource-oom-observed":
                require(report.get("oom_killed") is True, "resource failure lacks Docker OOM confirmation")
        results = report.get("results", {}).get("commands", {})
        if case == "wrong-result":
            require(results.get("compile-wrong-result", {}).get("exit_code") == 0,
                    "wrong assertion fixture did not compile")
            require(isinstance(results.get("run-wrong-result", {}).get("exit_code"), int) and
                    results.get("run-wrong-result", {}).get("exit_code") != 0 and
                    results.get("run-wrong-result", {}).get("signal") is None and
                    not results.get("run-wrong-result", {}).get("timed_out") and
                    "Assert_failure" in results.get("run-wrong-result", {}).get("stderr", ""),
                    "compiled wrong assertion did not produce its expected OCaml exception")
        if case == "syntax":
            require(report.get("fixture_hashes", {}).get("syntax.ml") == digest(b"let broken =\n"),
                    "syntax fixture bytes differ from the exact required source")
            diag = results.get("compile-syntax", {}).get("stderr", "")
            require(re.search(r"(?m)^Error: Syntax error", diag) is not None,
                    "syntax control lacks the specific OCaml syntax diagnostic")
        if case in ("timeout", "output", "resource"):
            control = results.get("run-" + case, {})
            require(results.get("compile-" + case, {}).get("exit_code") == 0,
                    f"{case} OCaml control did not compile through esy")
            if case == "timeout": require(control.get("reason") == "command-timeout", "compiled loop did not hit its deadline")
            if case == "output": require(control.get("reason") == "command-output-overflow", "compiled output did not overflow the command buffer")
            if case == "resource" and reason == "resource-allocation-failure-observed":
                require(control.get("reason") == "allocation-failure", "compiled allocation control lacks allocation failure evidence")
    return {"id": name, "case": case, "timeout_seconds": timeout,
            "expected_runner_status": "PASSED" if expect_pass else "FAILED",
            "actual": {"process": proc, "cli": cli, "report": report,
                       "independent_suite_cleanup": suite_cleanup}, "passed": True}


def main(argv: list[str] | None = None) -> int:
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--image", required=True)
    parser.add_argument("--expectations", required=True, type=Path)
    parser.add_argument("--output-dir", required=True, type=Path)
    parser.add_argument("--scenario", choices=("cleanup-results", "cleanup-finally-log-write"))
    args = parser.parse_args(argv)
    selected_runs, selected_missing = select_scenario(args.scenario)
    test_path = Path(__file__).resolve()
    proposal = test_path.parents[1]
    runner, dockerfile, fixture = proposal / "run_esy_reference.py", proposal / "Dockerfile.esy-runtime", proposal / "main.ml"
    candidate = test_path.parents[4]
    output = args.output_dir.resolve()
    manifest = {"schema_version": 1, "suite": "ESY_REFERENCE_REAL_DOCKER_E2E",
                "scope": args.scenario or "full-suite", "status": "FAILED",
                "image": args.image, "started_unix": time.time(), "scenarios": [], "failures": []}
    manifest_path = None
    try:
        require(IMAGE_RE.fullmatch(args.image) is not None, "--image must be immutable sha256 ID")
        require(not args.expectations.is_symlink(), "expectations cannot be a symlink")
        expectation_path = args.expectations.resolve(strict=True)
        require(expectation_path != candidate and candidate not in expectation_path.parents,
                "expectations must be outside the candidate tree")
        raw = bounded_read(expectation_path, MAX_EXPECTATIONS)
        expectation = check_expectations(raw, args.image, runner, dockerfile, fixture)
        require(output != candidate and candidate not in output.parents, "output directory must be outside candidate")
        require(not args.output_dir.is_symlink(), "output directory cannot be a symlink")
        require(not output.exists() or output.is_dir(), "output path must be a directory")
        output.mkdir(parents=True, exist_ok=True)
        require(not any(output.iterdir()), "output directory must be empty")
        manifest_path = output / "scenario-manifest.json"
        manifest.update({"expectations_path": str(expectation_path), "expectations_sha256": digest(raw),
                         "runner_sha256": digest(bounded_read(runner, 1024*1024)),
                         "test_sha256": digest(bounded_read(test_path, 1024*1024)),
                         "source_hashes": expectation["source_sha256"], "output_dir": str(output)})
        image_record = docker_inspect(args.image, proposal, output)
        require(image_record["layers"] == expectation["image_layers"], "actual image layers differ from captured expectations")
        manifest["image_evidence"] = image_record
        for name, case, timeout, expect_pass in selected_runs:
            try:
                manifest["scenarios"].append(run_case(name, case, timeout, expect_pass, runner,
                                                       args.image, expectation_path, output, proposal))
            except BaseException as exc:
                if isinstance(exc, ScenarioFailure):
                    row={"id":name,"case":case,"expected_runner_status":"PASSED" if expect_pass else "FAILED",
                         "passed":False,"failure":exc.payload}
                else:
                    row={"id":name,"case":case,"expected_runner_status":"PASSED" if expect_pass else "FAILED",
                         "passed":False,"failure":f"{type(exc).__name__}: {exc}"}
                manifest["scenarios"].append(row); manifest["failures"].append({"scenario":name,"reason":row["failure"]})
        for scenario, reason in selected_missing.items():
            row={"id":scenario,"passed":False,"evidence_status":"MISSING_FAILED","reason":reason}
            manifest["scenarios"].append(row); manifest["failures"].append({"scenario":scenario,"reason":reason})
        require(digest(bounded_read(runner,1024*1024)) == manifest["runner_sha256"], "runner source changed during suite")
        require(digest(bounded_read(test_path,1024*1024)) == manifest["test_sha256"], "E2E source changed during suite")
        require(digest(bounded_read(dockerfile,1024*1024)) == expectation["source_sha256"]["Dockerfile.esy-runtime"], "Dockerfile source changed during suite")
        require(digest(bounded_read(fixture,1024*1024)) == expectation["source_sha256"]["main.ml"], "fixture source changed during suite")
        if not manifest["failures"] and all(x.get("passed") is True for x in manifest["scenarios"]): manifest["status"]="PASSED"
    except BaseException as exc:
        manifest["failures"].append({"suite":f"{type(exc).__name__}: {exc}"})
    present = {row.get("id") for row in manifest["scenarios"]}
    for name, case, timeout, expect_pass in selected_runs:
        if name not in present:
            failure = "scenario setup failed before invocation"
            manifest["scenarios"].append({"id":name,"case":case,"timeout_seconds":timeout,
                "expected_runner_status":"PASSED" if expect_pass else "FAILED","passed":False,
                "evidence_status":"MISSING_FAILED","reason":failure})
            manifest["failures"].append({"scenario":name,"reason":failure})
    present = {row.get("id") for row in manifest["scenarios"]}
    for scenario, reason in selected_missing.items():
        if scenario not in present:
            manifest["scenarios"].append({"id":scenario,"passed":False,"evidence_status":"MISSING_FAILED","reason":reason})
            manifest["failures"].append({"scenario":scenario,"reason":reason})
    manifest["finished_unix"] = time.time()
    if manifest_path is None:
        # Validation failures before output ownership cannot safely publish there.
        sys.stdout.write(json.dumps({"status":"FAILED","manifest":None,"scope":manifest["scope"],"scenarios":manifest["scenarios"],"failures":manifest["failures"]},sort_keys=True)+"\n")
        return 1
    try:
        data=canonical(manifest)+b"\n"
        require(len(data)<=MAX_REPORT*8,"suite manifest exceeds evidence bound")
        h=immutable_write(manifest_path,data)
        sys.stdout.write(json.dumps({"status":manifest["status"],"manifest":str(manifest_path),"manifest_sha256":h,"failures":len(manifest["failures"])},sort_keys=True)+"\n")
    except BaseException as exc:
        sys.stdout.write(json.dumps({"status":"FAILED","manifest":str(manifest_path),"manifest_write_error":str(exc)},sort_keys=True)+"\n")
        return 1
    return 0 if manifest["status"]=="PASSED" else 1


if __name__ == "__main__":
    raise SystemExit(main())
