def digest(data):
    return hashlib.sha256(data).hexdigest()


def canonical(obj):
    return (json.dumps(obj, sort_keys=True, separators=(",", ":"), ensure_ascii=False) + "\n").encode()


def write_new(path, data):
    path.parent.mkdir(parents=True, exist_ok=True)
    fd = os.open(path, os.O_WRONLY | os.O_CREAT | os.O_EXCL, 0o600)
    with os.fdopen(fd, "wb") as f:
        f.write(data); f.flush(); os.fsync(f.fileno())
    dfd = os.open(path.parent, os.O_RDONLY)
    try: os.fsync(dfd)
    finally: os.close(dfd)


def strict_json(raw):
    def pairs(items):
        out = {}
        for k, v in items:
            if k in out: raise ValueError("duplicate JSON object key")
            out[k] = v
        return out
    return json.loads(raw.decode("utf-8", "strict"), object_pairs_hook=pairs,
                      parse_constant=lambda x: (_ for _ in ()).throw(ValueError("invalid constant")))

def file_manifest(root, *, permit_links=False):
    result = {}
    for base, dirs, files in os.walk(root, followlinks=False):
        bp = Path(base)
        for name in list(dirs):
            p = bp / name
            if p.is_symlink(): raise Incomplete(f"symlink forbidden: {p.relative_to(root)}")
        for name in files:
            p = bp / name; rel = p.relative_to(root).as_posix(); st = p.lstat()
            if not stat.S_ISREG(st.st_mode): raise Incomplete(f"special file forbidden: {rel}")
            if st.st_nlink > 1 and not permit_links: raise Incomplete(f"hardlinked file forbidden: {rel}")
            if st.st_size > POLICY["limits"]["file_bytes"]: raise Incomplete(f"file exceeds 2 MiB: {rel}")
            data = p.read_bytes()
            result[rel] = {"path": rel, "type": "file", "mode": stat.S_IMODE(st.st_mode),
                           "size": len(data), "sha256": digest(data)}
    return result

        print(f"INCOMPLETE: {e}", file=sys.stderr); return 2
    except Exception as e:
        print(f"INCOMPLETE: controller integrity failure: {e}", file=sys.stderr); return 3
    return 0

"""Standalone public-CLI checks in disposable authority/candidate fixtures."""
import hashlib

SOURCE = Path(__file__).resolve().parents[2] / "scripts/esy-repair/controller.py"


def sha(b): return hashlib.sha256(b).hexdigest()


def run(script, *args):
    return subprocess.run([sys.executable, str(script), *args], text=True,
                          stdout=subprocess.PIPE, stderr=subprocess.PIPE, timeout=15)


def fixture(base):
    root = base / "root"; candidate = root / "candidate"; auth = root / "controller-authority"
    candidate.mkdir(parents=True); auth.mkdir()
    (candidate / "README.md").write_bytes(b"frozen input\n")
    (candidate / "README.md").chmod(0o644)
    row = {"path": "README.md", "kind": "file", "mode": 0o644,
           "bytes": 13, "sha256": sha(b"frozen input\n")}
    (root / "baseline-manifest.json").write_text(json.dumps([row]))
    (root / "bootstrap").mkdir()
    (root / "bootstrap/pre-C01-manifest.json").write_text(json.dumps([row]))
    (root / "events.jsonl").write_text(json.dumps({"event": "resume-bootstrap", "base_active_seconds": 482.411}) + "\n")
    (root / "jobs/C01-core-001").mkdir(parents=True)
    (root / "jobs/C01-core-001/started.json").write_text(json.dumps({"role": "code", "wall_started": time.time()}))
    snap = auth / "controller.py"; shutil.copyfile(SOURCE, snap)
    return root, candidate, snap


def parser_case(cli, raw):
    code = ("import json,runpy,sys; m=runpy.run_path(sys.argv[1]); x=json.load(sys.stdin); "
            "\ntry: m['parse_provider'](bytes.fromhex(x['raw']),'r1','code','a'*64,'b'*64) "
            "\nexcept Exception as e: print(str(e)); sys.exit(2)\nprint('OK')")
    return subprocess.run([sys.executable, "-c", code, str(cli)], input=json.dumps({"raw": raw.hex()}),
                          text=True, stdout=subprocess.PIPE, stderr=subprocess.PIPE, timeout=15)


def test_protocol(cli):
    answer = {"schema_version": 1, "run_id": "r1", "role": "code", "revision_sha256": "a" * 64,
              "evidence_sha256": "b" * 64, "status": "complete", "summary": "ok", "changed_paths": []}
    ev = [{"type": "thread.started", "thread_id": "t1"}, {"type": "turn.started"},
          {"type": "item.completed", "item": {"id": "i1", "type": "agent_message", "text": json.dumps(answer)}},
          {"type": "turn.completed", "usage": {"input_tokens": 1, "cached_input_tokens": 0,
           "cache_write_input_tokens": 0, "output_tokens": 1, "reasoning_output_tokens": 0}}]
    valid = b"".join(json.dumps(x, separators=(",", ":")).encode() + b"\n" for x in ev)
    assert parser_case(cli, valid).stdout.strip() == "OK"
    bad = [b"{broken}\n", b'{"type":"thread.started","type":"thread.started"}\n', valid[:-1],
           valid + json.dumps(ev[-1]).encode() + b"\n"]
    stale = json.loads(json.dumps(ev)); stale[2]["item"]["text"] = json.dumps({**answer, "run_id": "old"})
    bad.append(b"".join(json.dumps(x).encode() + b"\n" for x in stale))
    failed = json.loads(json.dumps(ev)); failed[2] = {"type": "error", "message": "provider failure"}
    bad.append(b"".join(json.dumps(x).encode() + b"\n" for x in failed))
    for raw in bad:
        result = parser_case(cli, raw)
        assert result.returncode != 0, result


def main():
    with tempfile.TemporaryDirectory(prefix="esy-controller-e2e-") as td:
        root, candidate, cli = fixture(Path(td))
        r = run(cli, "init")
        assert r.returncode != 0 and "INCOMPLETE" in r.stderr and "initialized" in r.stderr, r
        r = run(cli, "print-handoff")
        assert r.returncode == 0 and '"state": "BOOTSTRAP"' in r.stdout, r
        r = run(cli, "resume")
        assert r.returncode != 0 and "lifecycle" in r.stderr, r
        r = run(cli, "launch", "--role", "plan")
        assert r.returncode != 0 and "dispatch" in r.stderr, r
        r = run(cli, "collect")
        assert r.returncode != 0 and "no candidate work" in r.stderr, r
        # Candidate edits remain separate from the installed executable.
        (candidate / "scripts").mkdir()
        (candidate / "scripts/esy-repair").mkdir()
        (candidate / "scripts/esy-repair/controller.py").write_text("candidate edit\n")
        r = run(cli, "collect")
        assert r.returncode != 0 and "controller.py" in r.stdout and "provider run evidence" in r.stderr, r
        first = json.loads(r.stdout)["run_id"]
        r = run(cli, "collect")
        assert r.returncode != 0 and json.loads(r.stdout)["run_id"] != first, r
        r = run(cli, "advance")
        assert r.returncode != 0 and "prerequisites" in r.stderr, r
        # A worker-created success marker is an out-of-card change and cannot
        # be used as acceptance evidence or mutate controller state.
        (candidate / "PASS").write_text("PASS\n")
        r = run(cli, "collect")
        assert r.returncode != 0 and "out-of-card" in r.stderr, r
        # Editing the candidate copy cannot change the installed authority.
        (candidate / "scripts/esy-repair/controller.py").write_text("raise SystemExit(0)\n")
        r = run(cli, "advance")
        assert r.returncode != 0 and "out-of-card" in r.stderr, r
        test_protocol(cli)
        lock = open(root / "controller-authority/controller.lock", "a+b")
        fcntl.flock(lock, fcntl.LOCK_EX | fcntl.LOCK_NB)
        try:
            r = run(cli, "collect")
            assert r.returncode != 0 and "exclusive lock" in r.stderr, r
        finally:
            fcntl.flock(lock, fcntl.LOCK_UN); lock.close()
    print("controller CLI E2E checks passed")


if __name__ == "__main__": main()
