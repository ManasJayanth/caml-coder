class Failure(Exception):
    pass


class RecordOverflow(Failure):
    def __init__(self, observed):
        self.observed = observed
        super().__init__("record-overflow")


def collect_records(path):
    chunks = []
    total = 0
    with open(path, "rb", buffering=0) as stream:
        while total < RECORD_LIMIT + 1:
            chunk = stream.read(RECORD_LIMIT + 1 - total)
            if not chunk:
                break
            chunks.append(chunk)
            total += len(chunk)
    raw = b"".join(chunks)
    if len(raw) > RECORD_LIMIT:
        raise RecordOverflow(len(raw))
    return raw


def validate_routing_delta(current, previous, expected_count, expected_record=None):
    result = {"expected_delta": expected_count, "observed_delta": 0,
              "routing_outcome": "match", "mismatch_reason": None,
              "observation": current[len(previous):][:512].decode("utf-8", "replace")
              if current.startswith(previous) else current[:512].decode("utf-8", "replace")}
    if not current.startswith(previous):
        result.update({"routing_outcome": "mismatch", "mismatch_reason": "snapshot-prefix"})
        return current, result
    delta = current[len(previous):]
    if expected_count == 0:
        if delta:
            result.update({"observed_delta": len(delta.split(bytes([10]))) - 1,
                           "routing_outcome": "mismatch", "mismatch_reason": "unexpected-bytes"})
        return current, result
    lines = delta.split(bytes([10]))
    result["observed_delta"] = len(lines) - 1 if lines[-1] == b"" else len(lines)
    if not delta.endswith(bytes([10])) or any(line == b"" for line in lines[:-1]) or lines[-1] != b"":
        result.update({"routing_outcome": "mismatch", "mismatch_reason": "blank-or-incomplete-line"})
        return current, result
    records = lines[:-1]
    if len(records) != expected_count:
        result.update({"routing_outcome": "mismatch", "mismatch_reason": "record-count"})
        return current, result
    try:
        decoded = [json.loads(line) for line in records]
    except (json.JSONDecodeError, UnicodeDecodeError):
        result.update({"routing_outcome": "mismatch", "mismatch_reason": "malformed-json"})
        return current, result
    if expected_record is not None and any(record != expected_record for record in decoded):
        result.update({"routing_outcome": "mismatch", "mismatch_reason": "content-mismatch"})
    return current, result


def absolute_nonsymlink(value, label):
    if not os.path.isabs(value):
        raise Failure(label + " must be absolute")
    path = Path(value)
    cur = Path(path.anchor)
    for part in path.parts[1:]:
        cur = cur / part
        try:
            mode = cur.lstat().st_mode
        except OSError as exc:
            raise Failure("invalid " + label) from exc
        if stat.S_ISLNK(mode):
            raise Failure(label + " has a symlink component")
    return path
