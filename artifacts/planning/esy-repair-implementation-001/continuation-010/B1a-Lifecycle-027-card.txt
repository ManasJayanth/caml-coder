def absolute_nonsymlink(value, label):
    if not os.path.isabs(value):
        raise Failure(label + " must be absolute")
    path = Path(value)
    cur = Path(path.anchor)
    for part in path.parts[1:]:
        cur = cur / part
        try:
            mode = cur.lstat().st_mode
        except OSError as exc:
            raise Failure("invalid " + label) from exc
        if stat.S_ISLNK(mode):
            raise Failure(label + " has a symlink component")
    return path


def validate(args):
    adapter = absolute_nonsymlink(args.adapter, "adapter")
    if not stat.S_ISREG(adapter.lstat().st_mode):
        raise Failure("adapter must be regular")
    sibling = adapter.parent / "codex-worker.sh"
    absolute_nonsymlink(str(sibling), "sibling adapter")
    if not stat.S_ISREG(sibling.lstat().st_mode):
        raise Failure("missing regular codex-worker.sh")
    output = absolute_nonsymlink(args.output_dir, "output directory")
    info = output.lstat()
    if not stat.S_ISDIR(info.st_mode) or info.st_uid != os.getuid() or stat.S_IMODE(info.st_mode) != 0o700:
        raise Failure("output directory must be owned, empty, and mode 0700")
    if os.listdir(output):
        raise Failure("output directory must be empty")
    return adapter, output


def write_bytes(path, data, mode=0o600):
    fd = os.open(path, os.O_WRONLY | os.O_CREAT | os.O_EXCL, mode)
    with os.fdopen(fd, "wb") as stream:
        stream.write(data)

def bounded_run(argv, workspace, home, output, records, timeout, lifecycle=None, scenario="matrix"):
    state = lifecycle if lifecycle is not None else {}
    state.update({"spawned_pid": None, "signal_attempts": [], "leader_reap": "not-reaped",
                  "pipe_closure": "pending", "cleanup": "pending", "failure": None})
    proc = None
    selector = None
    streams = {}
    phase = "not-spawned"
    deadline = time.monotonic() + timeout
    try:
        env = {"PATH": "/usr/bin:/bin", "HOME": str(home), "LANG": "C"}
        proc = subprocess.Popen(argv, cwd=str(workspace), env=env, stdin=subprocess.PIPE,
                               stdout=subprocess.PIPE, stderr=subprocess.PIPE, start_new_session=True,
                               close_fds=True)
        state["spawned_pid"] = proc.pid
        phase = "owned"
        streams = {proc.stdout: bytearray(), proc.stderr: bytearray()}
        if scenario == "lifecycle-setup-failure":
            raise Failure("deliberate selector setup failure")
        selector = selectors.DefaultSelector()
        for stream in streams:
            os.set_blocking(stream.fileno(), False)
            selector.register(stream, selectors.EVENT_READ)
        os.set_blocking(proc.stdin.fileno(), False)
        pending = memoryview(b"review fixture\\n")
        while selector.get_map() or pending:
            remaining = deadline - time.monotonic()
            if remaining <= 0:
                raise Failure("invocation timeout")
            if pending:
                try:
                    sent = os.write(proc.stdin.fileno(), pending)
                    pending = pending[sent:]
                except (BlockingIOError, BrokenPipeError):
                    pending = memoryview(b"") if isinstance(sys.exc_info()[1], BrokenPipeError) else pending
            if not pending and not proc.stdin.closed:
                proc.stdin.close()
            for key, _ in selector.select(min(remaining, 0.1)):
                chunk = os.read(key.fd, LIMIT + 1 - len(streams[key.fileobj]))
                if chunk:
                    streams[key.fileobj].extend(chunk)
                    if len(streams[key.fileobj]) > LIMIT:
                        raise Failure("stream overflow")
                else:
                    selector.unregister(key.fileobj)
                    key.fileobj.close()
        while True:
            # Mask interrupting signals before the syscall. If it reaps, phase is
            # changed before signals are unmasked; an interrupted/unknown wait fails closed.
            oldmask = signal.pthread_sigmask(signal.SIG_BLOCK, {signal.SIGINT, signal.SIGTERM})
            phase = "reap-uncertain"
            try:
                waited, raw_status = os.waitpid(proc.pid, os.WNOHANG)
                if waited == proc.pid:
                    phase = "reaped"
                    state["leader_reap"] = "reaped"
                else:
                    phase = "owned"
            finally:
                signal.pthread_sigmask(signal.SIG_SETMASK, oldmask)
            if phase == "reaped":
                break
            if phase != "owned":
                raise Failure("leader reap state uncertain")
            remaining = deadline - time.monotonic()
            if remaining <= 0:
                raise Failure("invocation timeout")
            time.sleep(min(0.02, remaining))
        proc.returncode = os.waitstatus_to_exitcode(raw_status)
        if scenario == "lifecycle-post-reap-exception":
            raise Failure("deliberate post-reap exception")
        if scenario == "lifecycle-post-reap-interrupt":
            raise KeyboardInterrupt("deliberate post-reap interruption")
        state["cleanup"] = "success"
        return bytes(streams[proc.stdout]), bytes(streams[proc.stderr]), proc.returncode
    except BaseException as exc:
        state["failure"] = type(exc).__name__ + ": " + str(exc)
        cleanup_deadline = time.monotonic() + 2.0
        cleanup_errors = []
        if proc is not None and phase == "owned":
            state["signal_attempts"].append({"phase": phase, "signal": "SIGKILL"})
            try:
                os.killpg(proc.pid, signal.SIGKILL)
            except ProcessLookupError:
                cleanup_errors.append("owned group disappeared")
            except OSError as err:
                cleanup_errors.append("group signal: " + str(err))
        elif phase == "reap-uncertain":
            cleanup_errors.append("leader ownership uncertain; refusing signal")
        # Drain without retaining data, then close every pipe under one deadline.
        for stream in ((proc.stdout, proc.stderr, proc.stdin) if proc is not None else ()):
            try:
                if stream in (proc.stdout, proc.stderr) and not stream.closed:
                    os.set_blocking(stream.fileno(), False)
                    while time.monotonic() < cleanup_deadline:
                        try:
                            if not os.read(stream.fileno(), LIMIT):
                                break
                        except BlockingIOError:
                            break
                stream.close()
            except OSError as err:
                cleanup_errors.append("pipe close: " + str(err))
        if proc is not None and phase == "owned":
            try:
                oldmask = signal.pthread_sigmask(signal.SIG_BLOCK, {signal.SIGINT, signal.SIGTERM})
                phase = "reap-uncertain"
                try:
                    waited, _ = os.waitpid(proc.pid, os.WNOHANG)
                    if waited == proc.pid:
                        phase = "reaped"
                        state["leader_reap"] = "reaped-during-cleanup"
                    else:
                        phase = "owned"
                finally:
                    signal.pthread_sigmask(signal.SIG_SETMASK, oldmask)
                if phase == "owned":
                    while time.monotonic() < cleanup_deadline:
                        time.sleep(0.01)
                        oldmask = signal.pthread_sigmask(signal.SIG_BLOCK, {signal.SIGINT, signal.SIGTERM})
                        phase = "reap-uncertain"
                        try:
                            waited, _ = os.waitpid(proc.pid, os.WNOHANG)
                            if waited == proc.pid:
                                phase = "reaped"
                                state["leader_reap"] = "reaped-during-cleanup"
                            else:
                                phase = "owned"
                        finally:
                            signal.pthread_sigmask(signal.SIG_SETMASK, oldmask)
                        if phase == "reaped":
                            break
                    if phase == "owned":
                        cleanup_errors.append("bounded reap expired")
            except OSError as err:
                cleanup_errors.append("reap: " + str(err))
        if selector is not None:
            try:
                selector.close()
            except Exception as err:
                cleanup_errors.append("selector close: " + str(err))
        state["pipe_closure"] = "failed" if cleanup_errors else "closed"
        state["cleanup"] = "failed" if cleanup_errors else "success"
        if cleanup_errors:
            raise Failure(state["failure"] + "; cleanup failed: " + "; ".join(cleanup_errors)) from exc
        raise
    finally:
        if selector is not None:
            selector.close()
        if proc is not None and state["pipe_closure"] == "pending":
            state["pipe_closure"] = "closed"


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--adapter", required=True)
    parser.add_argument("--output-dir", required=True)
    parser.add_argument("--scenario", choices=("matrix", "lifecycle-normal", "lifecycle-setup-failure",
                        "lifecycle-post-reap-exception", "lifecycle-post-reap-interrupt", "lifecycle-timeout",
                        "lifecycle-stdout-overflow", "lifecycle-stderr-overflow"), default="matrix")
    args = parser.parse_args()
    started = time.monotonic()
    try:
        adapter, output = validate(args)
        workspace, prompt, profile, fake, config, records, home = make_fixtures(output)
        if args.scenario != "matrix":
            probe = args.scenario
            response = {"stdout": CASES[0][1].hex(), "stderr": CASES[0][2].hex(), "status": 0}
            if probe == "lifecycle-timeout":
                response["sleep"] = INVOCATION_TIMEOUT + 1
            elif probe == "lifecycle-stdout-overflow":
                response["stdout"] = (b"x" * (LIMIT + 1)).hex()
            elif probe == "lifecycle-stderr-overflow":
                response["stderr"] = (b"x" * (LIMIT + 1)).hex()
            config.write_text(json.dumps(response), encoding="utf-8")
            state = {"scenario": probe}
            try:
                result = bounded_run(["/bin/bash", str(adapter), "review", str(workspace), str(prompt), str(profile), str(fake)],
                                    workspace, home, output, records, INVOCATION_TIMEOUT, state, probe)
                if result != (CASES[0][1], CASES[0][2], 0) or state["cleanup"] != "success":
                    raise Failure("lifecycle probe completion invalid")
            except BaseException:
                state["scenario"] = probe
                (output / "lifecycle.json").write_text(json.dumps(state, sort_keys=True), encoding="utf-8")
                raise
            (output / "lifecycle.json").write_text(json.dumps(state, sort_keys=True), encoding="utf-8")
            return 0
        rows = []
        for name, stdout, stderr, status in CASES:
            if time.monotonic() - started > SUITE_TIMEOUT:
                raise Failure("suite timeout")
            config.write_text(json.dumps({"stdout": stdout.hex(), "stderr": stderr.hex(), "status": status}), encoding="utf-8")
            argv = ["/bin/bash", str(adapter), "review", str(workspace), str(prompt), str(profile), str(fake)]
            actual_out, actual_err, actual_status = bounded_run(argv, workspace, home, output, records,
                                                                  min(INVOCATION_TIMEOUT, SUITE_TIMEOUT - (time.monotonic() - started)))
            rows.append((name, actual_out == stdout and actual_err == stderr and actual_status == status))
            if (actual_out, actual_err, actual_status) != (stdout, stderr, status):
                raise Failure("unexpected response for " + name)
        lines = records.read_bytes().splitlines()
        if len(lines) != len(CASES) or sum(map(len, lines)) > RECORD_LIMIT:
            raise Failure("record count or size invalid")
        expected = ["exec", "--disable", "apps", "--disable", "plugins", "--disable", "hooks", "--disable",
                    "multi_agent", "--disable", "shell_snapshot", "--disable", "shell_tool", "--ignore-user-config",
                    "--ephemeral", "--skip-git-repo-check", "--model", "gpt-6-astra", "--sandbox", "read-only",
                    "--cd", str(workspace), "--json", "-"]
        for line in lines:
            rec = json.loads(line)
            if rec != {"argv": expected, "cwd": str(workspace), "stdin": base64.b64encode(b"review fixture\n").decode("ascii")}:
                raise Failure("fake routing record mismatch")
        # Four arguments after the role is the valid review interface; exercise it
        # with real fixture paths. Never invoke the valid code branch.
        out, err, status = bounded_run(["/bin/bash", str(adapter), "review", str(workspace), str(prompt), str(profile), str(fake)],
                                       workspace, home, output, records,
                                       min(INVOCATION_TIMEOUT, SUITE_TIMEOUT - (time.monotonic() - started)))
        if (out, err, status) != (CASES[-1][1], CASES[-1][2], CASES[-1][3]):
            raise Failure("valid review arity failed")
        arities = [("review", n) for n in (0, 1, 2, 3, 6)] + [("code", 5), ("unknown", 4), ("unknown", 5)]
        for role, count in arities:
            params = [role] + ["unused"] * count
            out, err, status = bounded_run(["/bin/bash", str(adapter)] + params, workspace, home, output, records,
                                           min(INVOCATION_TIMEOUT, SUITE_TIMEOUT - (time.monotonic() - started)))
            if (out, err, status) != (b"", b"", 64):
                raise Failure("invalid arity was not rejected")
        if len(records.read_bytes().splitlines()) != len(CASES) + 1:
            raise Failure("invalid arity reached provider")
        print(json.dumps({"rows": [{"case": n, "status": "pass" if ok else "fail"} for n, ok in rows], "overall": "pass"}))
        return 0
    except Failure as exc:
        print(str(exc)[:DIAGNOSTIC_LIMIT], file=sys.stderr)
        return 1
    except Exception as exc:
        print((type(exc).__name__ + ": " + str(exc))[:DIAGNOSTIC_LIMIT], file=sys.stderr)
        return 1


if __name__ == "__main__":
    raise SystemExit(main())
