def absolute_nonsymlink(value, label):
    if not os.path.isabs(value):
        raise Failure(label + " must be absolute")
    path = Path(value)
    cur = Path(path.anchor)
    for part in path.parts[1:]:
        cur = cur / part
        try:
            mode = cur.lstat().st_mode
        except OSError as exc:
            raise Failure("invalid " + label) from exc
        if stat.S_ISLNK(mode):
            raise Failure(label + " has a symlink component")
    return path

def bounded_run(argv, workspace, home, output, records, timeout):
    env = {"PATH": "/usr/bin:/bin", "HOME": str(home), "LANG": "C"}
    proc = subprocess.Popen(argv, cwd=str(workspace), env=env, stdin=subprocess.PIPE,
                           stdout=subprocess.PIPE, stderr=subprocess.PIPE, start_new_session=True,
                           close_fds=True)
    selector = selectors.DefaultSelector()
    streams = {proc.stdout: bytearray(), proc.stderr: bytearray()}
    try:
        for stream in streams:
            os.set_blocking(stream.fileno(), False)
            selector.register(stream, selectors.EVENT_READ)
        os.set_blocking(proc.stdin.fileno(), False)
        pending = memoryview(b"review fixture\n")
        deadline = time.monotonic() + timeout
        while selector.get_map() or pending:
            remaining = deadline - time.monotonic()
            if remaining <= 0:
                raise Failure("invocation timeout")
            if pending:
                try:
                    sent = os.write(proc.stdin.fileno(), pending)
                    pending = pending[sent:]
                except BlockingIOError:
                    pass
                except BrokenPipeError:
                    pending = memoryview(b"")
            if not pending and proc.stdin is not None:
                proc.stdin.close()
            for key, _ in selector.select(min(remaining, 0.1)):
                chunk = os.read(key.fd, LIMIT + 1 - len(streams[key.fileobj]))
                if chunk:
                    streams[key.fileobj].extend(chunk)
                    if len(streams[key.fileobj]) > LIMIT:
                        raise Failure("stream overflow")
                else:
                    selector.unregister(key.fileobj)
                    key.fileobj.close()
        # waitpid(WNOHANG) retains ownership until exit is proven; unlike
        # Popen.poll()/wait(timeout), it cannot silently reap before cleanup.
        while True:
            waited, raw_status = os.waitpid(proc.pid, os.WNOHANG)
            if waited:
                proc.returncode = os.waitstatus_to_exitcode(raw_status)
                break
            if time.monotonic() >= deadline:
                raise Failure("invocation timeout")
            time.sleep(min(0.02, deadline - time.monotonic()))
        return bytes(streams[proc.stdout]), bytes(streams[proc.stderr]), proc.returncode
    except BaseException:
        try:
