#!/usr/bin/env python3
"""Independent subprocess-level client primitive for lifecycle probes."""
import argparse
import hashlib
import json
import os
import pathlib
import subprocess
import sys
import time

SCENARIOS = ('docker-metadata', 'fork-denial', 'timeout', 'caller-eof', 'output-overflow')

def main():
    parser = argparse.ArgumentParser(allow_abbrev=False)
    parser.add_argument('--supervisor', required=True)
    parser.add_argument('--output-dir', required=True)
    parser.add_argument('--scenario', required=True, choices=('client-primitive',))
    a = parser.parse_args()
    if not os.path.isabs(a.supervisor) or not os.path.isabs(a.output_dir) or os.path.lexists(a.output_dir):
        parser.error('supervisor and fresh output directory must be absolute')
    os.mkdir(a.output_dir, 0o700)
    os.chmod(a.output_dir, 0o700)
    root = pathlib.Path(a.output_dir)
    canary = subprocess.Popen([os.path.realpath(sys.executable), '-c', 'import time; time.sleep(60)'], stdin=subprocess.DEVNULL, stdout=subprocess.DEVNULL, stderr=subprocess.DEVNULL, close_fds=True, env={'PATH':'/usr/bin:/bin','LANG':'C','HOME':str(root)})
    rows=[]
    blocked=None
    try:
        for scenario in SCENARIOS:
            row={'scenario':scenario,'executed':False,'passed':False,'predicates':{},'failures':[]}
            rows.append(row)
            if blocked:
                row['failures'].append('not-run: '+blocked)
                continue
            state=root/('state-'+scenario)
            cmd=[os.path.realpath(sys.executable),os.path.realpath(a.supervisor),'probe','--scenario',scenario,'--state-directory',str(state)]
            proc=subprocess.Popen(cmd,stdin=subprocess.PIPE,stdout=subprocess.PIPE,stderr=subprocess.PIPE,close_fds=True,env={'PATH':'/usr/bin:/bin','LANG':'C','HOME':str(root)})
            end=time.monotonic()+5
            ready=None
            while time.monotonic()<end:
                rp=state/'ready.json'
                if rp.exists():
                    try: ready=json.loads(rp.read_text())
                    except (ValueError,OSError): pass
                    if ready and ready.get('scenario')==scenario and ready.get('supervisor_pid')==proc.pid and isinstance(ready.get('child_pid'),int): break
                if proc.poll() is not None: break
                time.sleep(.02)
            if scenario=='caller-eof' and ready:
                proc.stdin.close()
            else:
                # Keep the caller-owned write end open for every other scenario.
                pass
            try: out,err=proc.communicate(timeout=10)
            except subprocess.TimeoutExpired:
                proc.kill(); out,err=proc.communicate(); row['failures'].append('supervisor termination timeout')
            row['executed']=bool(ready)
            report=None
            try: report=json.loads((state/'report.json').read_text())
            except (OSError,ValueError): row['failures'].append('missing/invalid report')
            alive=canary.poll() is None
            row['predicates']['canary_alive']=alive
            if not alive: row['failures'].append('canary exited')
            valid=bool(ready and report and report.get('child_pid')==ready.get('child_pid') and report.get('reaped') is True and report.get('wait_status') is not None and report.get('probe_exit')==proc.returncode)
            row['predicates']['ready_report_waitpid_consistent']=valid
            row['predicates']['supervisor_terminated']=proc.poll() is not None
            if not valid: row['failures'].append('readiness/reap/exit receipt mismatch')
            if report:
                reason=report.get('reason'); streams=report.get('streams',{})
                if scenario=='docker-metadata':
                    ok=reason=='success' and report.get('probe_exit')==0 and all(v.get('eof') and v.get('observed_bytes')==v.get('retained_bytes')<=65536 for v in streams.values())
                    if ok:
                        try:
                            metadata=json.loads((state/'stdout.bin').read_bytes())
                            ok=isinstance(metadata,dict) and all(isinstance(metadata.get(k),str) and metadata[k] for k in ('Version','ApiVersion'))
                        except (ValueError,OSError): ok=False
                    row['predicates']['authentic_server_metadata']=ok
                elif scenario=='fork-denial':
                    ok=reason=='success' and report.get('probe_exit')==0 and b'FORK_DENIED:' in (state/'stdout.bin').read_bytes() and all(v.get('eof') for v in streams.values())
                    row['predicates']['explicit_fork_denial']=ok
                else:
                    expected={'timeout':'timeout','caller-eof':'caller-eof','output-overflow':'output-overflow'}[scenario]
                    ok=reason==expected and report.get('probe_exit')!=0 and report.get('reaped') is True
                    if scenario=='output-overflow': ok=ok and streams.get('stdout',{}).get('observed_bytes',0)>65536 and streams.get('stdout',{}).get('retained_bytes',65537)<=65536
                    row['predicates']['expected_failure_evidence']=ok
                if not row['predicates'].get('authentic_server_metadata',row['predicates'].get('explicit_fork_denial',row['predicates'].get('expected_failure_evidence',False))): row['failures'].append('scenario evidence predicate failed')
                if reason=='cleanup-unverified' or report.get('reason')=='executable-hash-mismatch': blocked=reason
            row['passed']=row['executed'] and not row['failures'] and all(row['predicates'].values())
        result={'scenario':'client-primitive','rows':rows,'success':len(rows)==5 and all(r['executed'] and r['passed'] for r in rows)}
        with open(root/'result.json','x') as f: json.dump(result,f,sort_keys=True); f.write('\n'); f.flush(); os.fsync(f.fileno())
        return 0 if result['success'] else 1
    finally:
        if canary.poll() is None:
            canary.terminate()
            try: canary.wait(timeout=2)
            except subprocess.TimeoutExpired: canary.kill(); canary.wait()

if __name__=='__main__': raise SystemExit(main())
