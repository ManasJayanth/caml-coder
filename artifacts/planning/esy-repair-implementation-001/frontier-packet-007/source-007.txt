def sha(data):
    return hashlib.sha256(data).hexdigest()


def canonical(value):
    return (json.dumps(value, sort_keys=True, separators=(",", ":"), ensure_ascii=False,
                       allow_nan=False) + "\n").encode("utf-8")


def put(path, data, mode=0o600):
    path = Path(path)
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_bytes(data if isinstance(data, bytes) else data.encode())
    path.chmod(mode)
    return path


def tree_manifest(root):
    rows = []
    for base, dirs, files in os.walk(root):
        base = Path(base)
        for name in sorted(dirs):
            p = base / name
            rows.append({"path": p.relative_to(root).as_posix() + "/", "type": "directory",
                         "mode": stat.S_IMODE(p.stat().st_mode)})
        for name in sorted(files):
            p = base / name
            data = p.read_bytes()
            rows.append({"path": p.relative_to(root).as_posix(), "type": "file",
                         "mode": stat.S_IMODE(p.stat().st_mode), "size": len(data), "sha256": sha(data)})
    return rows


def frozen_copy(src, dst):
    dst.mkdir(parents=True)
    for row in tree_manifest(src):
        rel = row["path"].rstrip("/")
        target = dst / rel
        if row["type"] == "directory":
            target.mkdir(parents=True, exist_ok=True)
        else:
            target.parent.mkdir(parents=True, exist_ok=True)
            shutil.copyfile(src / rel, target)
            target.chmod(row["mode"])
    for row in reversed(tree_manifest(src)):
        if row["type"] == "directory":
            (dst / row["path"].rstrip("/")).chmod(row["mode"])
    return tree_manifest(dst)


def event_rule(fields, types, item=None):
    out = {"fields": fields, "field_types": types, "nested_objects": {}}
    if item is not None:
        out["item"] = item
    return out


def provider_schema():
    events = {
        "thread.started": event_rule(["type", "thread_id"], {"type": "string", "thread_id": "string"}),
        "turn.started": event_rule(["type"], {"type": "string"}),
        "turn.completed": event_rule(["type"], {"type": "string"}),
    }
    for typ in ("command_execution", "file_change"):
        item_fields = ["id", "type", "status", "detail"]
        item_types = {"id": "string", "type": "string", "status": "string", "detail": "string"}
        item_rule = {"fields": item_fields, "field_types": item_types}
        for event_type in ("item.started", "item.updated", "item.completed"):
            key = event_type + ":" + typ
            events[key] = event_rule(["type", "item"], {"type": "string", "item": "object"}, item_rule)
    message_item = {"fields": ["id", "type", "text"],
                    "field_types": {"id": "string", "type": "string", "text": "string"}}
    events["item.completed:agent_message"] = event_rule(
        ["type", "item"], {"type": "string", "item": "object"}, message_item)
    return {"schema_version": 1, "events": events,
            "sequence": "thread-turn-items-terminal-v1", "final_response_item_type": "agent_message",
            "final_response_detection": {"kind": "strict_json_schema", "schema_version": 1}}

def cli(fixture, *args):
    return subprocess.run([str(fixture["interpreter"]), "-B", str(fixture["controller"]), *args],
                          text=True, stdout=subprocess.PIPE, stderr=subprocess.PIPE, timeout=45)


def main():
    passed = []
    failures = []
    blocked = []
    defect = None
    try:
        with tempfile.TemporaryDirectory(prefix="esy-controller-e2e-") as tmp:
            fixture = make_fixture(Path(tmp))
            root = str(fixture["root"])
            r = cli(fixture, "init", "--root", root)
            if r.returncode != 0 or json.loads(r.stdout).get("state") != "PLAN_REQUIRED":
                raise AssertionError(f"valid supervisor installation failed init: {r.stdout} {r.stderr}")
            passed.append("supervisor-installed schema, frozen adopted inputs, and PLAN_REQUIRED init")
            handoff = cli(fixture, "print-handoff", "--root", root)
            handoff_doc = json.loads(handoff.stdout)
            if handoff.returncode != 0 or handoff_doc.get("next_command") != [
                    "controller.py", "launch", "--root", root, "--role", "plan"]:
                raise AssertionError(f"initial handoff lacks actionable plan command: {handoff.stdout} {handoff.stderr}")
            if handoff_doc.get("coding_launches") != 7 or handoff_doc.get("active_seconds") != 482.411:
                raise AssertionError(f"historical accounting import mismatch: {handoff.stdout}")
            resumed = cli(fixture, "resume", "--root", root)
            if resumed.returncode != 0 or json.loads(resumed.stdout).get("state") != "PLAN_REQUIRED":
                raise AssertionError(f"resume did not preserve initialized state: {resumed.stdout} {resumed.stderr}")
            passed.append("resume and handoff preserve state, seven-launch accounting, and next action")
            for role in ("code", "review"):
                missing = cli(fixture, "launch", "--root", root, "--role", role)
                if missing.returncode == 0 or "--card is mandatory" not in missing.stderr:
                    raise AssertionError(f"{role} launch accepted missing card: {missing.stdout} {missing.stderr}")
            advance = cli(fixture, "advance", "--root", root)
            if advance.returncode == 0 or "advance unsupported from PLAN_REQUIRED" not in advance.stderr:
                raise AssertionError(f"advance accepted missing plan prerequisites: {advance.stdout} {advance.stderr}")
            passed.append("missing card and plan prerequisites fail explicitly through public CLI")

            # This is deliberately a failing acceptance assertion once the
            # plan provider has run: the installed controller records a phase
            # its own replay validator rejects.  Keep the valid launch attempt
            # intact so a controller repair can turn this into full lifecycle.
            launch = cli(fixture, "launch", "--root", root, "--role", "plan")
            if launch.returncode != 0:
                history_path = fixture["root"] / "history.jsonl"
                history = [json.loads(line) for line in history_path.read_text().splitlines()]
                jobs_root = fixture["root"] / "jobs"
                job_dirs = list(jobs_root.iterdir()) if jobs_root.is_dir() else []
                finished = json.loads((job_dirs[0] / "finished.json").read_text()) if job_dirs and (job_dirs[0] / "finished.json").exists() else {}
                provider_completed = (finished.get("failure") is None and finished.get("exit_status") == 0 and
                                      (job_dirs[0] / "stdout.jsonl").is_file() and
                                      (job_dirs[0] / "stdout.jsonl").stat().st_size > 0)
                if (provider_completed and len(history) >= 2 and history[1].get("phase") == "PLAN_RUNNING" and
                        "malformed history event fields" in launch.stderr):
                    defect = "controller writes PLAN_RUNNING, which replay rejects because PHASE omits it"
                    failures.append("successful plan/code/review lifecycle (PLAN_RUNNING replay defect)")
                else:
                    defect = ("plan provider/supervisor did not complete cleanly: " +
                              str(finished.get("failure") or launch.stderr.strip()))
                    failures.append("native plan launch/provider fixture failed before lifecycle")
            else:
                started = json.loads(launch.stdout.splitlines()[0])
                collect = cli(fixture, "collect", "--root", root, "--run-id", started["run_id"])
                if collect.returncode != 0:
                    defect = "plan launch returned success but public collect failed: " + collect.stderr.strip()
                    failures.append("successful plan collection and remaining lifecycle")
                else:
                    blocked.append("full code/review lifecycle is outside this bounded fixture run")
            print(json.dumps({"suite": "esy-repair public CLI E2E", "passed_supervised_core": passed,
                              "failed": failures, "blocked": blocked,
                              "controller_defect": defect,
                              "full_lifecycle_and_C2-C7": "INCOMPLETE",
                              "reviewed_source_hashes": {"controller.py": sha(SOURCE.read_bytes()),
                                                          "codex-worker.sh": sha(WRAPPER_SOURCE.read_bytes()),
                                                          "test_esy_repair_controller.py": sha(Path(__file__).read_bytes())}},
                             sort_keys=True))
            return 1 if failures else 0
    except RuntimeError as exc:
        blocked.append(str(exc))
        print(json.dumps({"suite": "esy-repair public CLI E2E", "passed_supervised_core": passed,
                          "failed": failures, "blocked": blocked,
                          "full_lifecycle_and_C2-C7": "INCOMPLETE"}, sort_keys=True))
        return 2


if __name__ == "__main__":
    raise SystemExit(main())

+        rel = row["path"].rstrip("/")
+        target = dst / rel
+        if row["type"] == "directory":
+            target.mkdir(parents=True, exist_ok=True)
+        else:
+            target.parent.mkdir(parents=True, exist_ok=True)
+            shutil.copyfile(src / rel, target)
+            target.chmod(row["mode"])
+    for row in reversed(tree_manifest(src)):
+        if row["type"] == "directory":
+            (dst / row["path"].rstrip("/")).chmod(row["mode"])
+    return tree_manifest(dst)
+

+def event_rule(fields, types, item=None):
+    out = {"fields": fields, "field_types": types, "nested_objects": {}}
+    if item is not None:
+        out["item"] = item
+    return out
+

+def provider_schema():
+    events = {
+        "thread.started": event_rule(["type", "thread_id"], {"type": "string", "thread_id": "string"}),
+        "turn.started": event_rule(["type"], {"type": "string"}),
+        "turn.completed": event_rule(["type"], {"type": "string"}),
+    }
+    for typ in ("command_execution", "file_change"):
+        item_fields = ["id", "type", "status", "detail"]
+        item_types = {"id": "string", "type": "string", "status": "string", "detail": "string"}
+        item_rule = {"fields": item_fields, "field_types": item_types}
+        for event_type in ("item.started", "item.updated", "item.completed"):
+            key = event_type + ":" + typ
+            events[key] = event_rule(["type", "item"], {"type": "string", "item": "object"}, item_rule)
+    message_item = {"fields": ["id", "type", "text"],
+                    "field_types": {"id": "string", "type": "string", "text": "string"}}
+    events["item.completed:agent_message"] = event_rule(
+        ["type", "item"], {"type": "string", "item": "object"}, message_item)
+    return {"schema_version": 1, "events": events,
+            "sequence": "thread-turn-items-terminal-v1", "final_response_item_type": "agent_message",
+            "final_response_detection": {"kind": "strict_json_schema", "schema_version": 1}}
+
