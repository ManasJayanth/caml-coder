#!/bin/bash
set -euo pipefail
case "$#" in
  4|5) ;;
  *) exit 64 ;;
esac
role=$1
case "$role" in
 review)
  [[ $# -eq 5 ]] || exit 64
  workspace=$2
  prompt=$3
  profile=$4
  codex_bin=$5
  script_dir=$(CDPATH= cd -- "$(dirname -- "${BASH_SOURCE[0]}")" && pwd)
  exec "$script_dir/codex-worker.sh" "$role" "$workspace" "$prompt" "$profile" "$codex_bin"
  ;;
 code)
  [[ $# -eq 4 ]] || exit 64
  workspace=$2
  prompt=$3
  profile=$4
  : "${CLAUDE_CODING_MODEL:?exact coding model required}"
  model=$CLAUDE_CODING_MODEL
  tools="Read,Edit,Write,Glob,Grep"
  permission=acceptEdits
  ;;
 *) exit 64 ;;
esac
: "${CLAUDE_BIN:?absolute executable required}"
cd "$workspace"
exec /usr/bin/sandbox-exec -f "$profile" "$CLAUDE_BIN" --print --model "$model" --output-format json --no-session-persistence --restricted --setting-sources "" --strict-mcp-config --mcp-config '{"mcpServers":{}}' --permission-mode "$permission" --tools "$tools" < "$prompt"
