class Failure(Exception):
    pass


class RecordOverflow(Failure):
    def __init__(self, observed):
        self.observed = observed
        super().__init__("record-overflow")


def collect_records(path):
    chunks = []
    total = 0
    with open(path, "rb", buffering=0) as stream:
        while total < RECORD_LIMIT + 1:
            chunk = stream.read(RECORD_LIMIT + 1 - total)
            if not chunk:
                break
            chunks.append(chunk)
            total += len(chunk)
    raw = b"".join(chunks)
    if len(raw) > RECORD_LIMIT:
        raise RecordOverflow(len(raw))
    return raw


def validate_routing_delta(current, previous, expected_count, expected_record=None):
    result = {"expected_delta": expected_count, "observed_delta": 0,
              "routing_outcome": "match", "mismatch_reason": None,
              "observation": current[len(previous):][:512].decode("utf-8", "replace")
              if current.startswith(previous) else current[:512].decode("utf-8", "replace")}
    if not current.startswith(previous):
        result.update({"routing_outcome": "mismatch", "mismatch_reason": "snapshot-prefix"})
        return current, result
    delta = current[len(previous):]
    if expected_count == 0:
        if delta:
            result.update({"observed_delta": len(delta.split(bytes([10]))) - 1,
                           "routing_outcome": "mismatch", "mismatch_reason": "unexpected-bytes"})
        return current, result
    lines = delta.split(bytes([10]))
    result["observed_delta"] = len(lines) - 1 if lines[-1] == b"" else len(lines)
    if not delta.endswith(bytes([10])) or any(line == b"" for line in lines[:-1]) or lines[-1] != b"":
        result.update({"routing_outcome": "mismatch", "mismatch_reason": "blank-or-incomplete-line"})
        return current, result
    records = lines[:-1]
    if len(records) != expected_count:
        result.update({"routing_outcome": "mismatch", "mismatch_reason": "record-count"})
        return current, result
    try:
        decoded = [json.loads(line) for line in records]
    except (json.JSONDecodeError, UnicodeDecodeError):
        result.update({"routing_outcome": "mismatch", "mismatch_reason": "malformed-json"})
        return current, result
    if expected_record is not None and any(record != expected_record for record in decoded):
        result.update({"routing_outcome": "mismatch", "mismatch_reason": "content-mismatch"})
    return current, result


def absolute_nonsymlink(value, label):
    if not os.path.isabs(value):
        raise Failure(label + " must be absolute")
    path = Path(value)
    cur = Path(path.anchor)
    for part in path.parts[1:]:
        cur = cur / part
        try:
            mode = cur.lstat().st_mode
        except OSError as exc:
            raise Failure("invalid " + label) from exc
        if stat.S_ISLNK(mode):
            raise Failure(label + " has a symlink component")
    return path


def validate(args):
    adapter = absolute_nonsymlink(args.adapter, "adapter")
    if not stat.S_ISREG(adapter.lstat().st_mode):
        raise Failure("adapter must be regular")
    sibling = adapter.parent / "codex-worker.sh"
    absolute_nonsymlink(str(sibling), "sibling adapter")
    if not stat.S_ISREG(sibling.lstat().st_mode):
        raise Failure("missing regular codex-worker.sh")
    output = absolute_nonsymlink(args.output_dir, "output directory")
    info = output.lstat()
    if not stat.S_ISDIR(info.st_mode) or info.st_uid != os.getuid() or stat.S_IMODE(info.st_mode) != 0o700:
        raise Failure("output directory must be owned, empty, and mode 0700")
    if os.listdir(output):
        raise Failure("output directory must be empty")
    return adapter, output


def write_bytes(path, data, mode=0o600):
    fd = os.open(path, os.O_WRONLY | os.O_CREAT | os.O_EXCL, mode)
    with os.fdopen(fd, "wb") as stream:
        stream.write(data)

def bounded_run(argv, workspace, home, output, records, suite_deadline, lifecycle=None, scenario="matrix"):
    state = lifecycle if lifecycle is not None else {}
    state.update({"spawned_pid": None, "first_signal": None,
                  "signal_attempt": {"intended": False, "attempted": False, "result": "not-needed", "phase": None},
                  "leader_reap": "not-reaped", "pipes": {"stdin": "absent", "stdout": "absent", "stderr": "absent"},
                  "selector": "absent", "cleanup_interrupted": False, "cleanup": "pending",
                  "cleanup_errors": [], "failure": None})
    proc = None
    selector = None
    streams = {}
    phase = "not-spawned"
    handlers = {}
    received = {"signal": 0, "cleanup": False}
    finalizing = False
    primary = None
    cleanup_deadline = None
    invocation_deadline = None

    def deferred_handler(signum, frame):
        if not received["signal"]:
            received["signal"] = signum
        if finalizing:
            received["cleanup"] = True

    def record_error(label, exc):
        state["cleanup_errors"].append(label + ": " + type(exc).__name__ + ": " + str(exc))

    def close_pipe(name, stream):
        if state["pipes"][name] in ("closed", "already-closed"):
            return
        if stream is None or stream.closed:
            state["pipes"][name] = "already-closed"
            return
        try:
            stream.close()
            state["pipes"][name] = "closed"
        except BaseException as exc:
            state["pipes"][name] = "failed/uncertain"
            record_error(name + " close", exc)

    def wait_once():
        nonlocal phase
        phase = "reap-uncertain"
        try:
            waited, raw = os.waitpid(proc.pid, os.WNOHANG)
        except BaseException:
            raise
        if waited == 0:
            phase = "owned"
            return False
        if waited != proc.pid:
            raise Failure("unexpected waitpid result")
        phase = "reaped"
        state["leader_reap"] = "reaped"
        proc.returncode = os.waitstatus_to_exitcode(raw)
        return True

    try:
        for signum in (signal.SIGINT, signal.SIGTERM):
            handlers[signum] = signal.signal(signum, deferred_handler)
        env = {"PATH": "/usr/bin:/bin", "HOME": str(home), "LANG": "C"}
        observed = time.monotonic()
        suite_remaining = suite_deadline - observed
        if suite_remaining <= 0:
            raise Failure("suite timeout")
        invocation_deadline = min(observed + INVOCATION_TIMEOUT, suite_deadline)
        proc = subprocess.Popen(argv, cwd=str(workspace), env=env, stdin=subprocess.PIPE,
                               stdout=subprocess.PIPE, stderr=subprocess.PIPE, start_new_session=True,
                               close_fds=True)
        phase = "owned"
        state["spawned_pid"] = proc.pid
        state["pipes"] = {"stdin": "pending", "stdout": "pending", "stderr": "pending"}
        streams = {proc.stdout: bytearray(), proc.stderr: bytearray()}
        if received["signal"]:
            raise Failure("interrupted after child acquisition")
        if scenario in ("lifecycle-setup-failure", "lifecycle-cleanup-interrupt"):
            raise Failure("deliberate selector setup failure")
        selector = selectors.DefaultSelector()
        state["selector"] = "open"
        for stream in streams:
            os.set_blocking(stream.fileno(), False)
            selector.register(stream, selectors.EVENT_READ)
        os.set_blocking(proc.stdin.fileno(), False)
        pending = memoryview(b"review fixture\n")
        while selector.get_map() or pending:
            if received["signal"]:
                raise Failure("interrupted during collection")
            remaining = invocation_deadline - time.monotonic()
            if remaining <= 0:
                raise Failure("invocation timeout")
            if pending:
                try:
                    sent = os.write(proc.stdin.fileno(), pending)
                    pending = pending[sent:]
                except BlockingIOError:
                    pass
                except BrokenPipeError:
                    pending = memoryview(b"")
            if not pending:
                close_pipe("stdin", proc.stdin)
            for key, _ in selector.select(min(remaining, 0.1)):
                chunk = os.read(key.fd, LIMIT + 1 - len(streams[key.fileobj]))
                if chunk:
                    streams[key.fileobj].extend(chunk)
                    if len(streams[key.fileobj]) > LIMIT:
                        raise Failure("stream overflow")
                else:
                    selector.unregister(key.fileobj)
                    close_pipe("stdout" if key.fileobj is proc.stdout else "stderr", key.fileobj)
            if received["signal"]:
                raise Failure("interrupted during collection")
        while True:
            if received["signal"]:
                raise Failure("interrupted during collection")
            if wait_once():
                break
            remaining = invocation_deadline - time.monotonic()
            if remaining <= 0:
                raise Failure("invocation timeout")
            time.sleep(min(0.02, remaining))
        if scenario == "lifecycle-post-reap-exception":
            raise Failure("deliberate post-reap exception")
        if scenario == "lifecycle-post-reap-interrupt":
            raise KeyboardInterrupt("deliberate post-reap interruption")
    except BaseException as exc:
        primary = exc
        state["failure"] = type(exc).__name__ + ": " + str(exc)

    # One finalizer owns all remaining obligations; signal handlers remain active.
    finalizing = True
    cleanup_deadline = time.monotonic() + 2.0
    if scenario == "lifecycle-cleanup-interrupt" and proc is not None:
        try:
            os.kill(os.getpid(), signal.SIGINT)
        except BaseException as exc:
            record_error("cleanup probe signal", exc)
    if primary is not None and proc is not None and phase in ("owned", "not-spawned") and phase != "not-spawned":
        state["signal_attempt"] = {"intended": True, "attempted": True, "result": "pending", "phase": phase}
        try:
            os.killpg(proc.pid, signal.SIGKILL)
            state["signal_attempt"]["result"] = "sent"
        except BaseException as exc:
            state["signal_attempt"]["result"] = "failed"
            record_error("group signal", exc)
    elif phase == "reap-uncertain":
        record_error("leader ownership", Failure("uncertain; refusing group signal"))

    for stream, name in ((proc.stdout, "stdout"), (proc.stderr, "stderr")) if proc is not None else ():
        if state["pipes"][name] not in ("closed", "already-closed"):
            try:
                os.set_blocking(stream.fileno(), False)
                while time.monotonic() < cleanup_deadline:
                    try:
                        if not os.read(stream.fileno(), LIMIT):
                            break
                    except BlockingIOError:
                        break
            except BaseException as exc:
                record_error(name + " drain", exc)
            close_pipe(name, stream)
    if proc is not None:
        close_pipe("stdin", proc.stdin)
        if phase in ("owned", "reap-uncertain") and phase == "owned":
            try:
                while True:
                    if wait_once():
                        break
                    remaining = cleanup_deadline - time.monotonic()
                    if remaining <= 0:
                        record_error("leader reap", Failure("bounded reap expired"))
                        break
                    time.sleep(min(0.01, remaining))
            except BaseException as exc:
                record_error("leader reap", exc)
        elif phase == "reap-uncertain":
            state["leader_reap"] = "uncertain"
    if selector is not None and state["selector"] == "open":
        try:
            selector.close()
            state["selector"] = "closed"
        except BaseException as exc:
            state["selector"] = "failed/uncertain"
            record_error("selector close", exc)
    for signum, previous in handlers.items():
        try:
            signal.signal(signum, previous)
        except BaseException as exc:
            record_error("signal handler restore", exc)
    state["cleanup_interrupted"] = bool(received["cleanup"])
    state["first_signal"] = received["signal"] or None
    if received["signal"] and not received["cleanup"]:
        if primary is None:
            primary = Failure("deferred signal " + str(received["signal"]))
            state["failure"] = str(primary)
    if received["cleanup"]:
        record_error("cleanup interruption", Failure("deferred signal observed"))
    all_pipes_closed = all(value in ("absent", "closed", "already-closed") for value in state["pipes"].values())
    reaped_or_absent = proc is None or phase == "reaped"
    clean = (not state["cleanup_errors"] and not state["cleanup_interrupted"] and all_pipes_closed
             and reaped_or_absent and state["selector"] in ("closed", "absent"))
    state["pipe_closure"] = "closed" if all_pipes_closed else "failed"
    state["cleanup"] = "success" if clean else "failed"
    if not clean and primary is None:
        primary = Failure("finalization failed")
        state["failure"] = str(primary)
    if primary is not None:
        if state["cleanup_errors"]:
            raise Failure(state["failure"] + "; cleanup failed: " + "; ".join(state["cleanup_errors"])) from primary
        raise primary
    return bytes(streams[proc.stdout]), bytes(streams[proc.stderr]), proc.returncode


def main():

    parser.add_argument("--adapter", required=True)

                        "deadline-suite-before-first", "deadline-suite-before-next",

                        "routing-duplicate", "diagnostic-ascii", "diagnostic-multibyte",

    args = parser.parse_args()
    started = time.monotonic()
    suite_deadline = started + SUITE_TIMEOUT
    try:

            raise KeyboardInterrupt()

        if args.scenario in ("deadline-suite-before-first", "deadline-suite-before-next"):

                        raise Failure("deadline control first invocation invalid")

        rows = []

+                    if observed != RECORD_LIMIT:
+                        raise Failure("exact record bound not observed")
+                    bounds_outcome = "accepted"
+                except RecordOverflow as exc:
+                    observed = exc.observed
+                    bounds_outcome = "rejected"
+                    if probe != "records-one-over":
+                        raise
+                if probe == "records-one-over":
+                    if observed != RECORD_LIMIT + 1 or bounds_outcome != "rejected":
+                        raise Failure("record-overflow boundary invalid")
+                    raise Failure("record-overflow")
+                if observed != RECORD_LIMIT or bounds_outcome != "accepted":
+                    raise Failure("exact record bound invalid")
+            except BaseException as exc:
+                original_failure = type(exc).__name__ + ": " + str(exc)
+                if isinstance(exc, RecordOverflow):
+                    observed = exc.observed
+                    bounds_outcome = "rejected"
+                elif observed is None and bounds_outcome == "not-evaluated":
+                    bounds_outcome = "failed"
+                receipt = {"scenario": probe, "observed_bytes": observed, "limit": RECORD_LIMIT,
+                           "bounds_outcome": bounds_outcome, "protocol_evaluated": False,
+                           "original_failure": original_failure, "invocation_lifecycle": state}
+                (output / "record-bound.json").write_text(json.dumps(receipt, sort_keys=True), encoding="utf-8")
+                raise
+            receipt = {"scenario": probe, "observed_bytes": observed, "limit": RECORD_LIMIT,
+                       "bounds_outcome": bounds_outcome, "protocol_evaluated": False,
+                       "original_failure": None, "invocation_lifecycle": state}
+            (output / "record-bound.json").write_text(json.dumps(receipt, sort_keys=True), encoding="utf-8")
+            return 0
+        if args.scenario not in ("matrix", "routing-missing", "routing-duplicate"):
+            probe = args.scenario
+            response = {"stdout": CASES[0][1].hex(), "stderr": CASES[0][2].hex(), "status": 0}
+            if probe == "lifecycle-timeout":
+                response["sleep"] = INVOCATION_TIMEOUT + 1
+            elif probe == "lifecycle-stdout-overflow":
+                response["stdout"] = (b"x" * (LIMIT + 1)).hex()
+            elif probe == "lifecycle-stderr-overflow":
+                response["stderr"] = (b"x" * (LIMIT + 1)).hex()
+            config.write_text(json.dumps(response), encoding="utf-8")
+            state = {"scenario": probe}
+            try:
+                result = bounded_run(["/bin/bash", str(adapter), "review", str(workspace), str(prompt), str(profile), str(fake)],
+                                    workspace, home, output, records, suite_deadline, state, probe)
+                if result != (CASES[0][1], CASES[0][2], 0) or state["cleanup"] != "success":
+                    raise Failure("lifecycle probe completion invalid")
+            except BaseException as exc:
+                state["scenario"] = probe
+                if state.get("failure") is None:
+                    state["failure"] = type(exc).__name__ + ": " + str(exc)
+                (output / "lifecycle.json").write_text(json.dumps(state, sort_keys=True), encoding="utf-8")
+                raise
+            (output / "lifecycle.json").write_text(json.dumps(state, sort_keys=True), encoding="utf-8")
+            return 0
+        rows = []
+        snapshot = b""
+        expected_argv = EXPECTED_ARGV.copy()
+        expected_argv[expected_argv.index("WORKSPACE")] = str(workspace)
+        expected_record = {"argv": expected_argv, "cwd": str(workspace),
+                           "stdin": base64.b64encode(b"review fixture\n").decode("ascii")}
+        control = args.scenario in ("routing-missing", "routing-duplicate")
+        lifecycle = {"scenario": args.scenario} if control else None
+        attempted = 0
+        for index, (name, stdout, stderr, status) in enumerate(CASES):
+            if time.monotonic() - started > SUITE_TIMEOUT:
+                raise Failure("suite timeout")
+            response = {"stdout": stdout.hex(), "stderr": stderr.hex(), "status": status}
+            if control and index == 0:
+                response["routing_mode"] = "missing" if args.scenario == "routing-missing" else "duplicate"
+            config.write_text(json.dumps(response), encoding="utf-8")
+            argv = ["/bin/bash", str(adapter), "review", str(workspace), str(prompt), str(profile), str(fake)]
+            actual_out = actual_err = None
+            actual_status = None
+            invocation_failure = None
+            if control:
+                attempted += 1
+            try:
+                actual_out, actual_err, actual_status = bounded_run(
+                    argv, workspace, home, output, records, suite_deadline, lifecycle,
+                    args.scenario if control else "matrix")
+            except BaseException as exc:
+                invocation_failure = type(exc).__name__ + ": " + str(exc)
+            current = collect_records(records)
+            _, routing = validate_routing_delta(current, snapshot, 1, expected_record)
+            rows.append({"case": name, **routing})
+            if control:
+                actual = {"stdout": actual_out[:512].decode("utf-8", "replace") if actual_out is not None else None,
+                          "stderr": actual_err[:512].decode("utf-8", "replace") if actual_err is not None else None,
+                          "status": actual_status}
+                receipt = {"scenario": args.scenario, "failing_row": name, **routing,
+                           "attempted_invocations": attempted, "invocation_lifecycle": lifecycle,
+                           "actual_observation": actual, "invocation_failure": invocation_failure}

+                        raise Failure("routing control response changed")
+                    raise Failure("routing mismatch for " + name)
+                raise Failure("routing control unexpectedly completed without mismatch")
+            if invocation_failure:
+                raise Failure(invocation_failure)
+            if routing["routing_outcome"] != "match":
+                raise Failure("routing mismatch for " + name)
+            snapshot = current
+            if (actual_out, actual_err, actual_status) != (stdout, stderr, status):
+                raise Failure("unexpected response for " + name)
+
+        arity_cases = [("empty-argv", [])]
+        arity_cases.extend(("review-total-" + str(count), ["review"] + ["unused"] * (count - 1))
+                           for count in (1, 2, 3, 4, 6))
+        arity_cases.append(("code-total-5", ["code"] + ["unused"] * 4))
+        arity_cases.extend(("unknown-total-" + str(count), ["unknown"] + ["unused"] * (count - 1))
+                           for count in (4, 5))
+        for name, params in arity_cases:

+            launch_argv = ["/bin/bash", str(adapter)] + adapter_argv

+            out, err, status = bounded_run(launch_argv, workspace, home, output, records, suite_deadline)
+            current = collect_records(records)
+            _, routing = validate_routing_delta(current, snapshot, 0)
+            rows.append({"case": name, "adapter_argv": adapter_argv,
+                         "total_argument_count": total_argument_count, "launch_argv": launch_argv, **routing})
+            if routing["routing_outcome"] != "match":
+                raise Failure("routing mismatch for invalid arity")
+            snapshot = current
+            if (status != 64 or out != b"" or err != b"" or routing["observed_delta"] != 0):
+                raise Failure("invalid arity was not rejected cleanly")
+        print(json.dumps({"rows": rows, "overall": "pass"}))
+        return 0
+    except Failure as exc:
+        sys.stderr.buffer.write(str(exc).encode("utf-8", "replace")[:DIAGNOSTIC_LIMIT - 1] + b"\n")
+        return 1
+    except KeyboardInterrupt:

+        return 130

+        message = type(exc).__name__ + ": " + str(exc)
+        sys.stderr.buffer.write(message.encode("utf-8", "replace")[:DIAGNOSTIC_LIMIT - 1] + b"\n")
+        return 1
